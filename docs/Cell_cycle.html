make_state_phase_df <- function(seu_obj) {
  meta <- seu_obj@meta.data
  
  meta %>%
    dplyr::filter(!is.na(donor_id),
           !is.na(cell_states),
           !is.na(Phase),
           !is.na(Region_x)) %>%
    # cells per donor × region × state × phase
    group_by(donor_id, Region_x, cell_states, Phase) %>%
    summarise(n_phase = n(), .groups = "drop") %>%
    # total cells per donor × region × state
    group_by(donor_id, Region_x, cell_states) %>%
    dplyr::mutate(n_total_state = sum(n_phase),
           pct_phase_in_state = n_phase / n_total_state * 100) %>%
    ungroup()
}

compare_state_vs_rest <- function(df, region) {
  df_region <- df %>% filter(Region_x == region)
  
  all_states <- levels(df_region$cell_states)
  all_phases <- levels(df_region$Phase)
  
  res_list <- list()
  
  for (ph in all_phases) {
    df_ph <- df_region %>% filter(Phase == ph)
    
    for (st in all_states) {
      # target state
      df_target <- df_ph %>%
        filter(cell_states == st) %>%
        dplyr::select(donor_id, pct_phase_in_state) %>%
        rename(target_pct = pct_phase_in_state)
      
      # "other" = all remaining states pooled
      df_other <- df_ph %>%
        filter(cell_states != st) %>%
        group_by(donor_id) %>%
        summarise(
          n_phase_other = sum(n_phase),
          n_total_other = sum(n_total_state),   # total cells in non-target states
          other_pct = (n_phase_other / n_total_other) * 100,
          .groups = "drop"
        )
      
      # merge; keep donors having both
      df_pair <- inner_join(df_target, df_other, by = "donor_id")
      
      if (nrow(df_pair) < 3) {
        # not enough donors with both target and other -> skip / NA
        res_list[[length(res_list) + 1]] <- data.frame(
          Region    = region,
          Phase     = ph,
          cell_states = st,
          n_donors  = nrow(df_pair),
          median_target_pct = NA_real_,
          median_other_pct  = NA_real_,
          p_value           = NA_real_
        )
        next
      }
      
      wtest <- suppressWarnings(
        wilcox.test(df_pair$target_pct, df_pair$other_pct, paired = TRUE)
      )
      
      res_list[[length(res_list) + 1]] <- data.frame(
        Region    = region,
        Phase     = ph,
        cell_states = st,
        n_donors  = nrow(df_pair),
        median_target_pct = median(df_pair$target_pct),
        median_other_pct  = median(df_pair$other_pct),
        p_value           = wtest$p.value
      )
    }
  }
  
  res <- bind_rows(res_list) %>%
    mutate(p_adj = p.adjust(p_value, method = "fdr"))
  
  return(res)
}

make_boxplot_phase_region <- function(df_state_phase, res_table,
                                      region, phase_label,
                                      sig_level = 0.05) {
  # data for this region & phase
  df_plot <- df_state_phase %>%
    filter(Region_x == region, Phase == phase_label)
  
  # significance for this region & phase
  res_sub <- res_table %>%
    filter(Region == region, Phase == phase_label) %>%
    mutate(
      label = case_when(
        is.na(p_adj) ~ "",
        p_adj < 0.001 ~ "***",
        p_adj < 0.01  ~ "**",
        p_adj < sig_level ~ "*",
        TRUE ~ ""
      )
    ) %>%
    filter(label != "")
  
  # y-position for asterisks: a bit above the max of each state
  if (nrow(res_sub) > 0) {
    label_df <- df_plot %>%
      group_by(cell_states) %>%
      summarise(y_max = max(pct_phase_in_state), .groups = "drop") %>%
      inner_join(res_sub, by = "cell_states") %>%
      mutate(y_pos = y_max * 1.05)
  } else {
    label_df <- NULL
  }
  
  p <- ggplot(df_plot, aes(x = cell_states, y = pct_phase_in_state, fill = cell_states)) +
    geom_boxplot(alpha = 0.7, outlier.shape = NA) +
    geom_jitter(width = 0.15, size = 1.8, alpha = 0.8) +
    labs(
      title = paste0(region, " – ", phase_label),
      x = "",
      y = "Percentage of cells in phase (per donor)"
    ) +
    theme_bw(base_size = 14) +
    theme(
      legend.position = "none",
      plot.title = element_text(face = "bold", size = 16),
      axis.text.x = element_text(angle = 45, hjust = 1)
    )
  
  if (!is.null(label_df) && nrow(label_df) > 0) {
    p <- p +
      geom_text(
        data = label_df,
        aes(x = cell_states, y = y_pos, label = label),
        vjust = 0,
        size = 5
      ) +
      expand_limits(y = max(label_df$y_pos) * 1.1)
  }
  
  return(p)
}
