simple_impute_for_pca <- function(mat) {
  out <- as.matrix(mat)
  global_floor <- suppressWarnings(stats::quantile(out, probs = 0.01, na.rm = TRUE))
  if (!is.finite(global_floor)) global_floor <- 0

  for (i in seq_len(nrow(out))) {
    miss <- is.na(out[i, ]) | !is.finite(out[i, ])
    if (!any(miss)) next

    obs <- out[i, !miss]
    if (length(obs) >= 3) {
      mu <- mean(obs, na.rm = TRUE)
      sdv <- stats::sd(obs, na.rm = TRUE)
      if (!is.finite(sdv) || sdv == 0) sdv <- 0.3
      out[i, miss] <- stats::rnorm(sum(miss), mean = mu - 1.8 * sdv, sd = max(0.05, 0.3 * sdv))
    } else {
      out[i, miss] <- global_floor
    }
  }

  out[!is.finite(out)] <- global_floor
  out
}

logit_prop <- function(x, eps = 0.001) {
  x <- pmin(pmax(x, eps), 1 - eps)
  log(x / (1 - x))
}

run_pca_from_matrix <- function(expr_mat, meta_df, assay_label) {
  sample_keep <- intersect(meta_df$Sample, colnames(expr_mat))

  meta_df <- meta_df %>%
    dplyr::filter(Sample %in% sample_keep) %>%
    dplyr::slice(match(sample_keep, Sample))

  expr_mat <- expr_mat[, meta_df$Sample, drop = FALSE]
  expr_pca <- simple_impute_for_pca(expr_mat)

  feature_sds <- matrixStats::rowSds(expr_pca, na.rm = TRUE)
  keep_feature <- is.finite(feature_sds) & feature_sds > 0
  expr_pca <- expr_pca[keep_feature, , drop = FALSE]

  pca <- prcomp(t(expr_pca), center = TRUE, scale. = TRUE)
  pvar <- 100 * (pca$sdev^2 / sum(pca$sdev^2))

  scores <- as.data.frame(pca$x) %>%
    tibble::rownames_to_column("Sample") %>%
    dplyr::left_join(meta_df, by = "Sample") %>%
    dplyr::mutate(assay = assay_label)

  list(
    pca = pca,
    pvar = pvar,
    scores = scores,
    expr_pca = expr_pca,
    meta = meta_df
  )
}

make_scree_plot <- function(pca_list, assay_label, n_pcs = 10) {
  n_pcs <- min(n_pcs, length(pca_list$pvar))

  df <- tibble(
    PC = factor(paste0("PC", seq_len(n_pcs)), levels = paste0("PC", seq_len(n_pcs))),
    variance = pca_list$pvar[seq_len(n_pcs)],
    cumulative = cumsum(pca_list$pvar[seq_len(n_pcs)])
  )

  ggplot(df, aes(x = PC, y = variance, group = 1)) +
    geom_col() +
    geom_line(aes(y = cumulative / max(cumulative) * max(variance))) +
    geom_point(aes(y = cumulative / max(cumulative) * max(variance))) +
    labs(
      title = paste0(assay_label, " scree / elbow plot"),
      x = NULL,
      y = "% variance explained"
    ) +
    theme_classic(base_size = 12)
}

extract_state_map <- function(file, family, prefix = NULL, state_col = NULL) {
  if (is.na(file)) {
    message("Skipping ", family, ": object not found.")
    return(NULL)
  }

  obj <- qs2::qs_read(file, nthreads = 32)

  if (is.null(state_col)) {
    state_col <- detect_state_col(obj)
  }

  meta <- obj@meta.data %>%
    tibble::rownames_to_column("cell") %>%
    dplyr::mutate(
      final_state = as.character(.data[[state_col]]),
      state_family = family
    ) %>%
    dplyr::filter(!is.na(final_state), final_state != "")

  if (!is.null(prefix)) {
    numeric_state <- grepl("^[0-9]+$", meta$final_state)
    meta$final_state[numeric_state] <- paste0(prefix, meta$final_state[numeric_state])
  }

  meta %>%
    dplyr::select(cell, final_state, state_family) %>%
    dplyr::distinct(cell, .keep_all = TRUE)
}

fit_pc_state_models <- function(scores_df, prop_df, state_info, assay = c("Olink", "MS"), pcs = 1:5,
                                min_samples_present = 10) {
  assay <- match.arg(assay)

  dat <- scores_df %>%
    dplyr::left_join(prop_df, by = "Sample")

  pc_names <- intersect(paste0("PC", pcs), colnames(dat))
  prop_cols_all <- grep("^prop_STATE_", colnames(dat), value = TRUE)

  prevalence <- purrr::map_dfr(prop_cols_all, function(prop_col) {
    final_state <- sub("^prop_STATE_", "", prop_col)

    tibble(
      final_state = final_state,
      prop_col = prop_col,
      n_present = sum(dat[[prop_col]] > 0, na.rm = TRUE),
      max_prop = max(dat[[prop_col]], na.rm = TRUE)
    )
  })

  prop_cols <- prevalence %>%
    dplyr::filter(n_present >= min_samples_present) %>%
    dplyr::pull(prop_col)

  logit_cols <- gsub("^prop_", "logit_prop_", prop_cols)

  covars <- c("Age_years")
  if (assay == "Olink") covars <- c(covars, "BCA_mg_ml")
  covars <- covars[covars %in% colnames(dat)]

  res <- purrr::map_dfr(pc_names, function(pc) {
    purrr::map2_dfr(prop_cols, logit_cols, function(prop_col, logit_col) {
      final_state <- sub("^prop_STATE_", "", prop_col)
      cols_needed <- c(pc, prop_col, logit_col, covars)

      df <- dat %>%
        dplyr::select(Sample, dplyr::all_of(cols_needed)) %>%
        tidyr::drop_na()

      if (nrow(df) < 5 || stats::sd(df[[logit_col]], na.rm = TRUE) == 0) {
        return(tibble(
          assay = assay,
          PC = pc,
          final_state = final_state,
          n = nrow(df),
          estimate = NA_real_,
          p_value = NA_real_,
          rho_prop = NA_real_,
          p_prop = NA_real_
        ))
      }

      form <- reformulate(termlabels = c(pc, covars), response = logit_col)
      fit <- lm(form, data = df)
      coef_tab <- summary(fit)$coefficients

      ct <- suppressWarnings(cor.test(df[[pc]], df[[prop_col]], method = "spearman"))

      tibble(
        assay = assay,
        PC = pc,
        final_state = final_state,
        n = nrow(df),
        estimate = coef_tab[pc, "Estimate"],
        p_value = coef_tab[pc, "Pr(>|t|)"],
        rho_prop = unname(ct$estimate),
        p_prop = ct$p.value
      )
    })
  })

  res %>%
    dplyr::left_join(state_info, by = "final_state") %>%
    dplyr::left_join(prevalence %>% dplyr::select(final_state, n_present, max_prop), by = "final_state") %>%
    dplyr::mutate(
      p_adj = p.adjust(p_value, method = "BH"),
      neg_log10_padj = -log10(p_adj),
      signed_neg_log10_padj = sign(estimate) * neg_log10_padj
    ) %>%
    dplyr::arrange(p_adj, p_value)
}

plot_focused_pc_state <- function(assay, pc, state) {
  obj <- get_pair_objects(assay)

  prop_col <- get_state_prop_col(state)
  logit_col <- get_state_logit_col(state)

  covars <- c("Age_years")
  if (assay == "Olink") covars <- c(covars, "BCA_mg_ml")
  covars <- covars[covars %in% colnames(obj$scores)]

  dat <- obj$scores %>%
    dplyr::left_join(all_state_props, by = "Sample") %>%
    dplyr::select(Sample, dplyr::all_of(c(pc, prop_col, logit_col, covars))) %>%
    tidyr::drop_na()

  if (nrow(dat) < 5) {
    return(ggplot() + annotate("text", x = 1, y = 1, label = "Too few samples") + theme_void())
  }

  form <- reformulate(termlabels = c(pc, covars), response = logit_col)
  fit <- lm(form, data = dat)
  coef_tab <- summary(fit)$coefficients

  beta <- coef_tab[pc, "Estimate"]
  pval <- coef_tab[pc, "Pr(>|t|)"]

  fdr <- obj$res %>%
    dplyr::filter(PC == pc, final_state == state) %>%
    dplyr::pull(p_adj)

  if (length(fdr) == 0) fdr <- NA_real_

  ct <- suppressWarnings(cor.test(dat[[pc]], dat[[prop_col]], method = "spearman"))

  ggplot(dat, aes(x = .data[[prop_col]], y = .data[[pc]], label = Sample)) +
    geom_point(size = 3) +
    geom_smooth(method = "lm", se = TRUE, linewidth = 0.5, color = "black") +
    ggrepel::geom_text_repel(size = 2.5, max.overlaps = 30) +
    annotate(
      "text",
      x = Inf, y = Inf,
      label = paste0(
        "beta(logit) = ", sprintf("%.3f", beta),
        "\np = ", signif(pval, 2),
        "\nFDR = ", signif(fdr, 2),
        "\nrho(prop) = ", round(unname(ct$estimate), 2),
        "\nn = ", nrow(dat)
      ),
      hjust = 1.1,
      vjust = 1.2,
      size = 4
    ) +
    labs(
      title = paste0(assay, " ", pc, " vs ", state, " proportion"),
      x = paste0(state, " proportion"),
      y = paste0(assay, " ", pc, " score")
    ) +
    theme_classic(base_size = 12)
}

make_focused_loading_heatmap <- function(assay, pc, state, top_n = 20) {
  obj <- get_pair_objects(assay)

  top_tbl <- obj$top_loadings %>%
    dplyr::filter(PC == pc) %>%
    dplyr::arrange(dplyr::desc(abs_loading)) %>%
    dplyr::slice_head(n = top_n)

  feat <- top_tbl$feature
  hm <- obj$expression[feat, , drop = FALSE]

  hm_z <- t(scale(t(hm)))
  hm_z[is.nan(hm_z)] <- 0
  hm_z[!is.finite(hm_z)] <- 0

  rownames(hm_z) <- top_tbl %>%
    dplyr::mutate(label = paste0(feature, " (", sprintf("%.3f", loading), ")")) %>%
    dplyr::pull(label)

  prop_col <- get_state_prop_col(state)

  ann <- all_state_props %>%
    dplyr::filter(Sample %in% colnames(hm_z)) %>%
    dplyr::slice(match(colnames(hm_z), Sample)) %>%
    dplyr::select(Sample, dplyr::all_of(prop_col)) %>%
    dplyr::rename(!!paste0("prop_", state) := dplyr::all_of(prop_col))

  # Order columns by selected cell-state proportion
  sample_order <- ann %>%
    dplyr::arrange(.data[[paste0("prop_", state)]]) %>%
    dplyr::pull(Sample)

  hm_z <- hm_z[, sample_order, drop = FALSE]

  ann <- ann %>%
    dplyr::slice(match(sample_order, Sample)) %>%
    tibble::column_to_rownames("Sample")

  # Cluster rows, but hide the row dendrogram
  row_hclust <- hclust(dist(hm_z), method = "complete")

  pheatmap(
    hm_z,
    annotation_col = ann,
    cluster_rows = row_hclust,
    treeheight_row = 0,
    cluster_cols = FALSE,
    show_rownames = TRUE,
    show_colnames = TRUE,
    fontsize_row = 8,
    fontsize_col = 8,
    border_color = NA,
    color = heat_cols,
    main = paste0(
      assay, " ", pc, " top ", top_n,
      " proteins - ", state, " proportion"
    )
  )
}

    genes,

    OrgDb = org.Hs.eg.db

    gene = genes_entrez,
    universe = universe_entrez,
    OrgDb = org.Hs.eg.db,
    ont = ont,
    pAdjustMethod = "BH",
    readable = TRUE
  )

  ego

read_marker_table_simple <- function(path) {
  if (length(path) == 0 || is.na(path) || !file.exists(path)) {
    stop("Marker file not found: ", path)
  }

  x <- read.csv(path, check.names = FALSE)

  # Fix empty column names from exported rownames
  empty_cols <- which(is.na(colnames(x)) | colnames(x) == "")
  if (length(empty_cols) > 0) {
    colnames(x)[empty_cols] <- paste0("unnamed_", empty_cols)
  }

  # If the first unnamed column is just rownames/index, drop it
  first_col <- colnames(x)[1]
  if (grepl("^unnamed_", first_col) || first_col %in% c("X", "...1")) {
    x <- x[, -1, drop = FALSE]
  }

  gene_col <- c("gene", "Gene", "genes", "symbol", "SYMBOL")
  gene_col <- gene_col[gene_col %in% colnames(x)][1]

  state_col <- c("cluster", "cell_state", "CellState", "state", "ident", "group")
  state_col <- state_col[state_col %in% colnames(x)][1]

  if (length(gene_col) == 0 || is.na(gene_col)) {
    stop(
      "Could not detect gene column in marker file: ", path,
      "\nAvailable columns are: ", paste(colnames(x), collapse = ", ")
    )
  }

  if (length(state_col) == 0 || is.na(state_col)) {
    stop(
      "Could not detect state/cluster column in marker file: ", path,
      "\nAvailable columns are: ", paste(colnames(x), collapse = ", ")
    )
  }

  x %>%
    dplyr::rename(
      marker_gene = dplyr::all_of(gene_col),
      marker_state = dplyr::all_of(state_col)
    ) %>%
    dplyr::mutate(
      marker_gene = as.character(marker_gene),
      marker_state = as.character(marker_state)
    )
}

get_top_pc_loading_simple <- function(assay, pc, top_n = 20) {
  loading_tbl <- if (assay == "MS") {
    ms_top_loadings
  } else {
    olink_top_loadings
  }

  loading_tbl %>%
    dplyr::filter(PC == pc) %>%
    dplyr::arrange(dplyr::desc(abs_loading)) %>%
    dplyr::slice_head(n = top_n) %>%
    dplyr::mutate(
      loading_rank = dplyr::row_number(),
      loading_gene = as.character(feature)
    )
}

make_pc_marker_overlap_simple <- function(assay, pc, cell_state, marker_file, top_n = 20) {
  top_tbl <- get_top_pc_loading_simple(assay, pc, top_n = top_n)

  marker_tbl <- read_marker_table_simple(marker_file) %>%
    dplyr::filter(marker_state == cell_state) %>%
    dplyr::distinct(marker_gene, .keep_all = TRUE)

  marker_cols <- intersect(
    c("marker_gene", "marker_state", "p_val", "avg_log2FC", "avg_logFC", "pct.1", "pct.2", "p_val_adj"),
    colnames(marker_tbl)
  )

  full_tbl <- top_tbl %>%
    dplyr::left_join(
      marker_tbl %>% dplyr::select(dplyr::all_of(marker_cols)),
      by = c("loading_gene" = "marker_gene")
    ) %>%
    dplyr::mutate(
      assay = assay,
      PC = pc,
      tested_state = cell_state,
      is_marker_of_state = !is.na(marker_state)
    ) %>%
    dplyr::select(
      assay, PC, tested_state,
      loading_rank, loading_gene, loading, abs_loading,
      is_marker_of_state,
      dplyr::everything()
    )

  summary_tbl <- tibble::tibble(
    assay = assay,
    PC = pc,
    tested_state = cell_state,
    top_n = top_n,
    n_top_pc_proteins = nrow(top_tbl),
    n_state_markers = nrow(marker_tbl),
    n_overlap = sum(full_tbl$is_marker_of_state),
    overlap_genes = paste(full_tbl$loading_gene[full_tbl$is_marker_of_state], collapse = "; ")
  )

  list(summary = summary_tbl, full = full_tbl, overlap = full_tbl %>% dplyr::filter(is_marker_of_state))
}
