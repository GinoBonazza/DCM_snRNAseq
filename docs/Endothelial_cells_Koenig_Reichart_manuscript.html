wrap_2lines <- function(x, width = 35) {
  vapply(x, function(s) {
    s <- as.character(s)
    if (nchar(s) <= width) return(s)
    cut <- stringr::str_locate_all(substr(s, 1, width), " ")[[1]]
    cut <- if (nrow(cut) == 0) width else max(cut[, 1])
    line1 <- stringr::str_trim(substr(s, 1, cut))
    rest  <- stringr::str_trim(substr(s, cut + 1, nchar(s)))
    if (nchar(rest) > width) rest <- paste0(substr(rest, 1, width - 1), "…")
    paste(line1, rest, sep = "\n")
  }, FUN.VALUE = character(1))
}


format_p_label <- function(p) {
  if (length(p) == 0 || is.na(p)) return("NA")
  dplyr::case_when(
    p < 0.0001 ~ "****",
    p < 0.001  ~ "***",
    p < 0.01   ~ "**",
    p < 0.05   ~ "*",
    TRUE ~ sprintf("%.4f", p)
  )
}

  if (length(p) == 0 || is.na(p)) return("plain")

  )
}

  gsea_down <- gsea_obj
  gsea_down@result <- gsea_res %>% dplyr::filter(NES < 0, p.adjust < 0.05)

  plot_list_emap <- list()

  if (nrow(gsea_up@result) > 1) {
    plot_list_emap[["up"]] <- emapplot(
      pairwise_termsim(gsea_up),
      showCategory = min(show_n, nrow(gsea_up@result)),
      size_category = 0.7,
      min_edge = 0.25
    ) +
      ggtitle("Activated terms") +
      theme(plot.title = element_text(face = "bold", hjust = 0.5))
  }

  if (nrow(gsea_down@result) > 1) {
    plot_list_emap[["down"]] <- emapplot(
      pairwise_termsim(gsea_down),
      showCategory = min(show_n, nrow(gsea_down@result)),
      size_category = 0.7,
      min_edge = 0.25
    ) +
      ggtitle("Suppressed terms") +
      theme(plot.title = element_text(face = "bold", hjust = 0.5))
  }

  if (length(plot_list_emap) == 2) {
    return(plot_list_emap[["up"]] + plot_list_emap[["down"]])
  } else if (length(plot_list_emap) == 1) {
    return(plot_list_emap[[1]])
  } else {
    return(NULL)
  }
}

make_gsea_barplot <- function(gsea_obj, label_width = 40, top_n = 10, base_size = 11) {
  if (is.null(gsea_obj) || nrow(gsea_obj@result) == 0) return(NULL)

  gsea_bar_df <- gsea_obj@result %>%
    as_tibble() %>%
    dplyr::filter(!is.na(NES), !is.na(p.adjust), p.adjust < 0.05)

  if (nrow(gsea_bar_df) == 0) return(NULL)

  gsea_bar_df <- gsea_bar_df %>%
    dplyr::mutate(score = -log10(p.adjust) * sign(NES))

  gsea_bar_df_pos <- gsea_bar_df %>%
    dplyr::filter(NES > 0) %>%
    dplyr::arrange(desc(NES)) %>%
    dplyr::slice_head(n = top_n)

  gsea_bar_df_neg <- gsea_bar_df %>%
    dplyr::filter(NES < 0) %>%
    dplyr::arrange(NES) %>%
    dplyr::slice_head(n = top_n)

  gsea_bar_df_plot <- dplyr::bind_rows(gsea_bar_df_pos, gsea_bar_df_neg)

  if (nrow(gsea_bar_df_plot) == 0) return(NULL)

  gsea_bar_df_plot$Description <- factor(
    gsea_bar_df_plot$Description,
    levels = gsea_bar_df_plot$Description[order(gsea_bar_df_plot$NES)]
  )

  L <- unname(quantile(abs(gsea_bar_df_plot$score), probs = 0.80, na.rm = TRUE))
  if (!is.finite(L) || L == 0) L <- max(abs(gsea_bar_df_plot$score), na.rm = TRUE)

  ggplot(gsea_bar_df_plot, aes(x = Description, y = NES, fill = score)) +
    geom_col(width = 0.9) +
    scale_fill_gradient2(
      low = "#004C8C",
      mid = "white",
      high = "#A63A2A",
      midpoint = 0,
      limits = c(-L, L),
      oob = scales::squish,
      name = expression(-log[10](adj.~p) %*% sign(NES))
    ) +
    scale_x_discrete(labels = function(x) wrap_2lines(x, width = label_width)) +
    coord_flip() +
    labs(x = NULL, y = "Normalized enrichment score (NES)") +
    theme_bw(base_size = base_size) +
    theme(
      text = element_text(color = "black"),
      axis.text = element_text(color = "black"),
      axis.title = element_text(color = "black", size = 13),
      plot.title = element_blank(),
      axis.text.y = element_text(size = 11),
      legend.position = "bottom",
      legend.direction = "horizontal",
      legend.title = element_text(size = 9),
      legend.text = element_text(size = 9)
    ) +
    guides(fill = guide_colorbar(
      barwidth = unit(6, "cm"),
      barheight = unit(0.3, "cm"),
      title.position = "top",
      title.hjust = 0.5
    ))
}

add_module_score_clean <- function(object, features, score_name, assay = NULL, nbin = 24, ctrl = 100, seed = 1) {
  if (is.null(assay)) {
    assay <- Seurat::DefaultAssay(object)
  }

  features_present <- intersect(features, rownames(object[[assay]]))
  if (length(features_present) == 0) {
    warning(paste0("No genes present for ", score_name, ". Filling with NA."))
    object[[score_name]] <- NA_real_
    return(object)
  }

  object <- Seurat::AddModuleScore(
    object = object,
    features = list(features_present),
    assay = assay,
    name = paste0(score_name, "_tmp"),
    nbin = nbin,
    ctrl = ctrl,
    seed = seed
  )

  tmp_col <- paste0(score_name, "_tmp1")
  object[[score_name]] <- object[[tmp_col]]
  object[[tmp_col]] <- NULL

  object
}

set_plot_layer_alpha <- function(p, alpha_value = 0.7) {
  for (i in seq_along(p$layers)) {
    geom_name <- class(p$layers[[i]]$geom)[1]
    if (geom_name %in% c("GeomPoint", "GeomJitter", "GeomScattermore")) {
      p$layers[[i]]$aes_params$alpha <- alpha_value
      if (!is.null(p$layers[[i]]$geom_params)) {
        p$layers[[i]]$geom_params$alpha <- alpha_value
      }
    }
  }
  p
}

plot_hvhf_by_subset <- function(df, value_col, title_text, ylab_text, facet_nrow = 2) {
  df <- df %>%
    dplyr::filter(!is.na(.data[[value_col]])) %>%
    dplyr::mutate(
      cell_state = factor(cell_state, levels = subset_levels),
      Group = factor(Group, levels = c("H", "HF"))
    )

  p_df <- lapply(subset_levels, function(cs) {
    df_sub <- df %>% dplyr::filter(cell_state == cs)
    if (nrow(df_sub) == 0 || dplyr::n_distinct(df_sub$Group) < 2) {

    }

    )

        dplyr::filter(term == "GroupHF") %>%
        dplyr::pull(p.value)

      NA_real_
    }

    data.frame(cell_state = cs, p_value = ifelse(length(p_val) == 0, NA_real_, p_val))

      y_rng = y_max - y_min,
      y_rng = ifelse(!is.finite(y_rng) | y_rng == 0, 1, y_rng),

      y_text = y_max + 0.16 * y_rng
    ) %>%

    dplyr::mutate(

    geom_segment(

      inherit.aes = FALSE
    ) +
    geom_text(
      data = anno_df,
      aes(x = 1.5, y = y_text, label = p_label, fontface = p_fontface),
      inherit.aes = FALSE,
      size = 4.2
    ) +
    scale_fill_manual(values = group_cols) +
    facet_wrap(~cell_state, scales = "free_y", nrow = facet_nrow) +
    labs(title = title_text, x = NULL, y = ylab_text) +

      plot.title = element_text(face = "bold", hjust = 0.5),

    )

format_p_label <- function(p) {
  if (is.na(p)) return("NA")
  if (p < 0.001) return("***")
  if (p < 0.01) return("**")
  if (p < 0.05) return("*")
  sprintf("p=%.2f", p)
}

format_p_fontface <- function(p) {
  ifelse(!is.na(p) && p < 0.05, "bold", "plain")
}

safe_colmean_score <- function(gene_vec, score_mat, ids) {
  genes_here <- intersect(gene_vec, rownames(score_mat))
  if (length(genes_here) == 0) {
    return(rep(NA_real_, length(ids)))
  }
  out <- colMeans(score_mat[genes_here, , drop = FALSE])
  as.numeric(out[ids])
}

plot_compact_score_by_subset <- function(df, value_col, title_text, ylab_text) {
  df <- df %>%
    dplyr::filter(!is.na(.data[[value_col]])) %>%
    dplyr::mutate(
      cell_state = factor(cell_state, levels = subset_levels),
      Group = factor(Group, levels = c("H", "HF"))
    )

  pd  <- position_dodge(width = 0.75)
  pjd <- position_jitterdodge(jitter.width = 0.12, dodge.width = 0.75, seed = 123)

  p_df <- lapply(subset_levels, function(cs) {
    df_sub <- df %>% dplyr::filter(cell_state == cs)
    if (nrow(df_sub) == 0 || dplyr::n_distinct(df_sub$Group) < 2) {
      return(data.frame(cell_state = cs, p_value = NA_real_))
    }

    fit <- tryCatch(
      lm(stats::as.formula(paste0(value_col, " ~ Group + Age + Sex + Dataset")), data = df_sub),
      error = function(e) NULL
    )

    p_val <- if (!is.null(fit)) {
      broom::tidy(fit) %>%
        dplyr::filter(term == "GroupHF") %>%
        dplyr::pull(p.value)
    } else {
      NA_real_
    }

    data.frame(cell_state = cs, p_value = ifelse(length(p_val) == 0, NA_real_, p_val))
  }) %>% dplyr::bind_rows()

  y_annot <- df %>%
    dplyr::group_by(cell_state) %>%
    dplyr::summarise(y = max(.data[[value_col]], na.rm = TRUE) + 0.35, .groups = "drop") %>%
    dplyr::left_join(p_df, by = "cell_state") %>%
    dplyr::mutate(
      p_label = vapply(p_value, format_p_label, character(1)),
      p_fontface = vapply(p_value, format_p_fontface, character(1))
    )

  ggplot(df, aes(x = cell_state, y = .data[[value_col]], fill = Group)) +
    geom_boxplot(
      outlier.shape = NA,
      alpha = 0.8,
      width = 0.7,
      position = pd
    ) +
    geom_point(
      position = pjd,
      size = 0.8,
      alpha = 0.65,
      shape = 16
    ) +
    geom_segment(
      data = y_annot,
      aes(
        x = as.numeric(cell_state) - 0.18,
        xend = as.numeric(cell_state) + 0.18,
        y = y,
        yend = y
      ),
      inherit.aes = FALSE,
      linewidth = 0.4
    ) +
    geom_text(
      data = y_annot,
      aes(
        x = cell_state,
        y = y + 0.10,
        label = p_label,
        fontface = p_fontface
      ),
      inherit.aes = FALSE,
      size = 3.2
    ) +
    scale_fill_manual(values = group_cols) +
    labs(title = title_text, x = NULL, y = ylab_text) +
    coord_cartesian(ylim = c(min(df[[value_col]], na.rm = TRUE), max(y_annot$y + 0.18, na.rm = TRUE))) +
    theme_classic(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", hjust = 0.5),
      axis.text.x = element_text(angle = 45, hjust = 1, face = "bold"),
      axis.text.y = element_text(face = "bold"),
      axis.title = element_text(face = "bold"),
      legend.title = element_blank()
    )
}

compute_subset_gene_stats <- function(df_program) {
  res <- purrr::map_dfr(sort(unique(df_program$cell_state)), function(cs) {
    purrr::map_dfr(sort(unique(df_program$gene)), function(g) {
      df_sub <- df_program %>% dplyr::filter(cell_state == cs, gene == g)

      if (nrow(df_sub) == 0 || dplyr::n_distinct(df_sub$Group) < 2) {
        return(data.frame(
          cell_state = cs,
          gene = g,
          estimate = NA_real_,
          p_value = NA_real_
        ))
      }

      fit <- tryCatch(
        lm(logCPM ~ Group + Age + Sex + Dataset, data = df_sub),
        error = function(e) NULL
      )

      if (is.null(fit)) {
        return(data.frame(
          cell_state = cs,
          gene = g,
          estimate = NA_real_,
          p_value = NA_real_
        ))
      }

      tt <- broom::tidy(fit) %>% dplyr::filter(term == "GroupHF")
      if (nrow(tt) == 0) {
        return(data.frame(
          cell_state = cs,
          gene = g,
          estimate = NA_real_,
          p_value = NA_real_
        ))

safe_colmean_score <- function(gene_vec, score_mat, ids) {
  genes_here <- intersect(gene_vec, rownames(score_mat))
  if (length(genes_here) == 0) {
    return(rep(NA_real_, length(ids)))
  }
  out <- colMeans(score_mat[genes_here, , drop = FALSE])
  as.numeric(out[ids])
}

format_p_label <- function(p) {
  if (is.na(p)) return("NA")
  if (p < 0.001) return("***")
  if (p < 0.01) return("**")
  if (p < 0.05) return("*")
  "ns"
}

format_p_fontface <- function(p) {
  ifelse(!is.na(p) & p < 0.05, "bold", "plain")
}

plot_hvhf_compact <- function(df, value_col, title_text, ylab_text) {
  pd <- position_dodge(width = 0.75)
  pjd <- position_jitterdodge(jitter.width = 0.12, dodge.width = 0.75, seed = 123)

  df <- df %>%
    dplyr::filter(!is.na(.data[[value_col]])) %>%
    dplyr::mutate(
      cell_state = factor(cell_state, levels = subset_levels),
      Group = factor(Group, levels = c("H", "HF"))
    )

  p_df <- lapply(subset_levels, function(cs) {
    df_sub <- df %>% dplyr::filter(cell_state == cs)
    if (nrow(df_sub) == 0 || dplyr::n_distinct(df_sub$Group) < 2) {
      return(data.frame(cell_state = cs, p_value = NA_real_))
    }

    fit <- tryCatch(
      lm(stats::as.formula(paste0(value_col, " ~ Group + Age + Sex + Dataset")), data = df_sub),
      error = function(e) NULL
    )

    p_val <- if (!is.null(fit)) {
      broom::tidy(fit) %>%
        dplyr::filter(term == "GroupHF") %>%
        dplyr::pull(p.value)
    } else {
      NA_real_
    }

    data.frame(cell_state = cs, p_value = ifelse(length(p_val) == 0, NA_real_, p_val))
  }) %>% dplyr::bind_rows()

  y_annot <- df %>%
    dplyr::group_by(cell_state) %>%
    dplyr::summarise(y = max(.data[[value_col]], na.rm = TRUE), .groups = "drop") %>%
    dplyr::left_join(p_df, by = "cell_state") %>%
    dplyr::mutate(
      y = y + 0.08 * diff(range(df[[value_col]], na.rm = TRUE)),
      label_y = y + 0.04 * diff(range(df[[value_col]], na.rm = TRUE)),
      p_label = vapply(p_value, format_p_label, character(1)),
      p_fontface = vapply(p_value, format_p_fontface, character(1))
    )

  ggplot(df, aes(x = cell_state, y = .data[[value_col]], fill = Group)) +
    geom_boxplot(
      outlier.shape = NA,
      alpha = 0.8,
      width = 0.7,
      position = pd
    ) +
    geom_point(
      position = pjd,
      size = 0.8,
      alpha = 0.65,
      shape = 16
    ) +
    geom_segment(
      data = y_annot,
      aes(
        x = as.numeric(cell_state) - 0.18,
        xend = as.numeric(cell_state) + 0.18,
        y = y,
        yend = y
      ),
      inherit.aes = FALSE,
      linewidth = 0.4
    ) +
    geom_text(
      data = y_annot,
      aes(
        x = cell_state,
        y = label_y,
        label = p_label,
        fontface = p_fontface
      ),
      inherit.aes = FALSE,
      size = 3.2
    ) +
    scale_fill_manual(values = group_cols) +
    labs(title = title_text, x = NULL, y = ylab_text) +
    coord_cartesian(clip = "off") +
    theme_classic(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", hjust = 0.5),
      axis.text.x = element_text(angle = 45, hjust = 1, face = "bold"),
      axis.text.y = element_text(face = "bold"),
      axis.title = element_text(face = "bold"),
      legend.title = element_blank(),
      legend.position = "top",
      plot.margin = margin(5.5, 20, 5.5, 5.5)
    )
}

make_subset_gene_stats <- function(df_program) {
  res <- lapply(split(df_program, interaction(df_program$cell_state, df_program$gene, drop = TRUE)), function(dd) {
    if (nrow(dd) == 0 || dplyr::n_distinct(dd$Group) < 2) return(NULL)

    fit <- tryCatch(
      lm(logCPM ~ Group + Age + Sex + Dataset, data = dd),
      error = function(e) NULL
    )
    if (is.null(fit)) return(NULL)

    tt <- broom::tidy(fit) %>%
      dplyr::filter(term == "GroupHF")

    if (nrow(tt) == 0) return(NULL)

    data.frame(
      cell_state = as.character(dd$cell_state[1]),
      gene = as.character(dd$gene[1]),
      estimate = tt$estimate[1],
      p_value = tt$p.value[1],
      stringsAsFactors = FALSE
    )
  }) %>% dplyr::bind_rows()

  if (is.null(res) || nrow(res) == 0) return(res)

  res %>%
    dplyr::mutate(
      padj = p.adjust(p_value, method = "fdr"),
      neglog10p = -log10(p_value),
      sig = padj < 0.05,
      cell_state = factor(cell_state, levels = rev(subset_levels))
    )
}
