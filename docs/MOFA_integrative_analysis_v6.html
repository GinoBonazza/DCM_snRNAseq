Overview

This workflow fits two independent MOFA models.

  1. Integrated proteomics model: Olink Reveal and LC-MS/MS are integrated in patients measured by both platforms. Factors are tested against continuous clinical parameters and between compensated and decompensated RV states. Factor 1 is interpreted by Gene Ontology gene set enrichment analysis of the complete ranked weights.

  2. Matched three-view model: pseudobulk snRNA-seq, Olink and LC-MS/MS are integrated in patients measured in all three modalities. The analysis distinguishes modality-specific from shared molecular variation. Factors are tested against RV state, continuous clinical parameters and histological fibrosis. Factor 2 is interpreted separately in the transcriptomic and proteomic layers by Gene Ontology gene set enrichment analysis.

Proteomic matrices and retained DCM samples are imported from the preceding proteomics quality-control workflow. BCA concentration is removed from the Olink view before model fitting. MOFA training is unsupervised; RV state and clinical variables are used only in downstream association models. Age is included as a covariate.

RV states are defined as:

  • Not overloaded: mPAP <= 20 mmHg;
  • Compensated: mPAP > 20 mmHg and TAPSE/sPAP > 0.36;
  • Decompensated: mPAP > 20 mmHg and TAPSE/sPAP <= 0.36.

Patients with missing mPAP receive NA for RV_group. Patients with mPAP > 20 mmHg and missing TAPSE/sPAP also receive NA.

standardise_mofa_view_name <- function(view) {
  dplyr::case_when(
    view == "RNA_pseudobulk" ~ "RNA",
    view == "Olink" ~ "OLINK",
    view == "LCMS" ~ "MS",
    TRUE ~ as.character(view)
  )
}


clean_mofa_feature_name <- function(
    feature
) {
  stringr::str_remove(
    as.character(feature),
    "_(RNA_pseudobulk|RNA|Olink|OLINK|LCMS|MS)$"
  )
}


format_p_value <- function(
    p_value,
    prefix = "P"
) {
  dplyr::case_when(
    is.na(p_value) ~
      paste0(prefix, " = NA"),
    p_value < 0.001 ~
      paste0(prefix, " < 0.001"),
    TRUE ~
      sprintf(
        "%s = %.3f",
        prefix,
        p_value
      )
  )
}


prepare_top_factor_loadings <- function(
    weight_table,
    factor_name = "Factor1",
    n_each_direction = 20
) {
  factor_weights <- weight_table %>%
    dplyr::filter(
      Factor == factor_name,
      is.finite(Weight)
    ) %>%
    dplyr::mutate(
      Modality =
        standardise_mofa_view_name(
          View
        ),
      Feature_clean =
        clean_mofa_feature_name(
          Feature
        ),
      Feature_label = paste(
        Modality,
        Feature_clean,
        sep = "_"
      )
    )

  positive_weights <- factor_weights %>%
    dplyr::filter(
      Weight > 0
    ) %>%
    dplyr::slice_max(
      order_by = Weight,
      n = n_each_direction,
      with_ties = FALSE
    )

  negative_weights <- factor_weights %>%
    dplyr::filter(
      Weight < 0
    ) %>%
    dplyr::slice_min(
      order_by = Weight,
      n = n_each_direction,
      with_ties = FALSE
    )

  dplyr::bind_rows(
    negative_weights,
    positive_weights
  ) %>%
    dplyr::arrange(
      Weight
    ) %>%
    dplyr::mutate(
      Feature_label = factor(
        Feature_label,
        levels = unique(
          Feature_label
        )
      )
    )
}

scale_mofa_weights_within_view <- function(
    weight_table,
    factor_table
) {
  weight_table %>%
    dplyr::filter(
      Factor %in%
        factor_table$Factor,
      is.finite(Weight)
    ) %>%
    dplyr::mutate(
      Feature_original =
        as.character(Feature),
      Feature =
        clean_mofa_feature_name(
          Feature
        )
    ) %>%
    dplyr::inner_join(
      factor_table,
      by = "Factor"
    ) %>%
    dplyr::group_by(
      Factor,
      View
    ) %>%
    dplyr::mutate(
      Oriented_weight =
        Weight * orientation,
      maximum_absolute_weight =
        max(
          abs(Oriented_weight),
          na.rm = TRUE
        ),
      Scaled_weight =
        dplyr::if_else(
          maximum_absolute_weight > 0,
          Oriented_weight /
            maximum_absolute_weight,
          0
        ),
      Absolute_scaled_weight =
        abs(Scaled_weight)
    ) %>%
    dplyr::ungroup()
}


prepare_mofa_ranked_gene_list <- function(
    weight_table,
    factor_name,
    orientation
) {
  ranked_symbols <-
    scale_mofa_weights_within_view(
      weight_table =
        weight_table,
      factor_table =
        tibble::tibble(
          Factor = factor_name,
          orientation = orientation,
          interpretation =
            factor_name
        )
    ) %>%
    dplyr::group_by(
      Factor,
      Feature
    ) %>%
    dplyr::summarise(
      Ranked_weight =
        mean(
          Scaled_weight,
          na.rm = TRUE
        ),
      .groups = "drop"
    ) %>%
    dplyr::filter(
      is.finite(
        Ranked_weight
      ),
      Ranked_weight != 0
    )

  symbol_to_entrez <- bitr(
    ranked_symbols$Feature,
    fromType = "SYMBOL",
    toType = "ENTREZID",
    OrgDb = org.Hs.eg.db
  ) %>%
    dplyr::distinct(
      SYMBOL,
      ENTREZID
    ) %>%
    dplyr::mutate(
      ENTREZID =
        as.character(
          ENTREZID
        )
    )

  ranked_entrez <-
    ranked_symbols %>%
    dplyr::inner_join(
      symbol_to_entrez,
      by = c(
        "Feature" =
          "SYMBOL"
      )
    ) %>%
    dplyr::group_by(
      ENTREZID
    ) %>%
    dplyr::slice_max(
      order_by =
        abs(Ranked_weight),
      n = 1,
      with_ties = FALSE
    ) %>%
    dplyr::ungroup() %>%
    dplyr::arrange(
      dplyr::desc(
        Ranked_weight
      )
    )

  gene_list <-
    ranked_entrez$Ranked_weight

  names(gene_list) <-
    ranked_entrez$ENTREZID

  sort(
    gene_list,
    decreasing = TRUE
  )
}


prefix_go_descriptions <- function(
    enrichment_result
) {
  if (
    is.null(enrichment_result) ||
    nrow(
      enrichment_result@result
    ) == 0L ||
    !"ONTOLOGY" %in%
      colnames(
        enrichment_result@result
      )
  ) {
    return(enrichment_result)
  }

  already_prefixed <-
    stringr::str_detect(
      enrichment_result@result$
        Description,
      "^\\[(BP|CC|MF)\\] "
    )

  enrichment_result@result$
    Description[
      !already_prefixed
    ] <- paste0(
      "[",
      enrichment_result@result$
        ONTOLOGY[
          !already_prefixed
        ],
      "] ",
      enrichment_result@result$
        Description[
          !already_prefixed
        ]
    )

  enrichment_result
}


run_mofa_gsea <- function(
    weight_table,
    factor_table
) {
  gsea_objects <- list()
  ranked_lists <- list()

  for (
    factor_index in
    seq_len(
      nrow(factor_table)
    )
  ) {
    factor_name <-
      factor_table$Factor[
        factor_index
      ]

    orientation <-
      factor_table$orientation[
        factor_index
      ]

    gene_list <-
      prepare_mofa_ranked_gene_list(
        weight_table =
          weight_table,
        factor_name =
          factor_name,
        orientation =
          orientation
      )

    ranked_lists[[
      factor_name
    ]] <- gene_list

    if (
      length(gene_list) <
        gsea_min_ranked_genes
    ) {
      next
    }

    go_result <- tryCatch(
      gseGO(
        geneList = gene_list,
        ont = "ALL",
        OrgDb = org.Hs.eg.db,
        keyType = "ENTREZID",
        exponent = 1,
        minGSSize = 5,
        maxGSSize = 500,
        eps = 1e-10,
        pvalueCutoff = 1,
        pAdjustMethod = "BH",
        verbose = FALSE,
        seed = TRUE,
        by = "fgsea"
      ),
      error = function(e) {
        warning(
          "GO GSEA failed for ",
          factor_name,
          ": ",
          conditionMessage(e)
        )
        NULL
      }
    )

    go_result <-
      prefix_go_descriptions(
        go_result
      )

    gsea_objects[[
      paste(
        factor_name,
        "GO_ALL",
        sep = "__"
      )
    ]] <- go_result
  }

  list(
    results =
      extract_gsea_results(
        gsea_objects
      ),
    objects =
      gsea_objects,
    ranked_lists =
      ranked_lists
  )
}


extract_gsea_results <- function(
    object_list
) {
  output <- tibble::tibble()

  for (
    object_name in
    names(object_list)
  ) {
    current_object <-
      object_list[[
        object_name
      ]]

    if (
      is.null(current_object) ||
      nrow(
        as.data.frame(
          current_object
        )
      ) == 0L
    ) {
      next
    }

    name_parts <-
      strsplit(
        object_name,
        "__",
        fixed = TRUE
      )[[1]]

    current_table <-
      as.data.frame(
        current_object
      ) %>%
      tibble::as_tibble() %>%
      dplyr::mutate(
        Factor =
          name_parts[1],
        Direction =
          dplyr::if_else(
            NES >= 0,
            "Positive",
            "Negative"
          ),
        database =
          name_parts[2]
      )

    output <-
      dplyr::bind_rows(
        output,
        current_table
      )
  }

  output
}


subset_enrichment_object <- function(
    enrichment_result,
    selected_ids
) {
  reduced_result <-
    enrichment_result

  reduced_result@result <-
    enrichment_result@result[
      enrichment_result@result$ID %in%
        selected_ids,
      ,
      drop = FALSE
    ]

  if (
    "geneSets" %in%
      methods::slotNames(
        reduced_result
      )
  ) {
    reduced_result@geneSets <-
      enrichment_result@geneSets[
        intersect(
          selected_ids,
          names(
            enrichment_result@geneSets
          )
        )
      ]
  }

  reduced_result
}


simplify_go_all_result <- function(
    enrichment_result,
    cutoff =
      enrichment_simplify_cutoff
) {
  if (
    is.null(enrichment_result) ||
    nrow(
      enrichment_result@result
    ) < 2L
  ) {
    return(enrichment_result)
  }

  result_table <-
    as.data.frame(
      enrichment_result
    ) %>%
    dplyr::filter(
      is.finite(p.adjust)
    ) %>%
    dplyr::arrange(
      p.adjust
    )

  selected_table <-
    result_table %>%
    dplyr::filter(
      p.adjust <
        enrichment_simplify_padj_cutoff
    )

  if (nrow(selected_table) < 2L) {
    selected_table <-
      result_table %>%
      dplyr::slice_head(
        n = min(
          enrichment_simplify_max_terms,
          nrow(result_table)
        )
      )
  }

  if (
    nrow(selected_table) >
      enrichment_simplify_max_terms
  ) {
    selected_table <-
      selected_table %>%
      dplyr::slice_head(
        n =
          enrichment_simplify_max_terms
      )
  }

  selected_object <-
    subset_enrichment_object(
      enrichment_result =
        enrichment_result,
      selected_ids =
        selected_table$ID
    )

  selected_result_table <-
    selected_object@result

  if (
    !"ONTOLOGY" %in%
      colnames(
        selected_result_table
      )
  ) {
    return(
      tryCatch(
        clusterProfiler::simplify(
          selected_object,
          cutoff = cutoff,
          by = "p.adjust",
          select_fun = min,
          measure = "Wang"
        ),
        error = function(e) {
          warning(
            "GO simplification failed: ",
            conditionMessage(e)
          )
          selected_object
        }
      )
    )
  }

  ontology_parts <- list()

  for (
    current_ontology in
    c(
      "BP",
      "CC",
      "MF"
    )
  ) {
    current_ids <-
      selected_result_table$ID[
        selected_result_table$
          ONTOLOGY ==
          current_ontology
      ]

    if (length(current_ids) == 0L) {
      next
    }

    current_object <-
      subset_enrichment_object(
        enrichment_result =
          selected_object,
        selected_ids =
          current_ids
      )

    if (
      "ontology" %in%
        methods::slotNames(
          current_object
        )
    ) {
      current_object@ontology <-
        current_ontology
    }

    if (
      "setType" %in%
        methods::slotNames(
          current_object
        )
    ) {
      current_object@setType <-
        current_ontology
    }

    ontology_parts[[
      current_ontology
    ]] <- tryCatch(
      clusterProfiler::simplify(
        current_object,
        cutoff = cutoff,
        by = "p.adjust",
        select_fun = min,
        measure = "Wang"
      ),
      error = function(e) {
        warning(
          "GO simplification failed for ",
          current_ontology,
          ": ",
          conditionMessage(e)
        )
        current_object
      }
    )
  }

  if (length(ontology_parts) == 0L) {
    return(selected_object)
  }

  combined_object <-
    selected_object

  combined_object@result <-
    do.call(
      rbind,
      lapply(
        ontology_parts,
        function(x) {
          x@result
        }
      )
    )

  if (
    "geneSets" %in%
      methods::slotNames(
        combined_object
      )
  ) {
    combined_object@geneSets <-
      unlist(
        lapply(
          ontology_parts,
          function(x) {
            x@geneSets
          }
        ),
        recursive = FALSE
      )
  }

  if (
    "ontology" %in%
      methods::slotNames(
        combined_object
      )
  ) {
    combined_object@ontology <-
      "ALL"
  }

  if (
    "setType" %in%
      methods::slotNames(
        combined_object
      )
  ) {
    combined_object@setType <-
      "ALL"
  }

  combined_object
}


simplify_enrichment_objects <- function(
    object_list
) {
  lapply(
    names(object_list),
    function(object_name) {
      current_object <-
        object_list[[
          object_name
        ]]

      if (
        stringr::str_ends(
          object_name,
          "GO_ALL"
        )
      ) {
        simplify_go_all_result(
          current_object,
          cutoff =
            enrichment_simplify_cutoff
        )
      } else {
        current_object
      }
    }
  ) %>%
    stats::setNames(
      names(object_list)
    )
}


fit_factor_fibrosis_models <- function(
    factor_scores,
    factor_names
) {
  required_columns <- c(
    "Factor",
    "Factor_value",
    "Fibrosis_percent",
    "Age_years"
  )

  missing_columns <- setdiff(
    required_columns,
    colnames(factor_scores)
  )

  if (length(missing_columns) > 0L) {
    stop(
      "Missing fibrosis-analysis columns: ",
      paste(
        missing_columns,
        collapse = ", "
      )
    )
  }

  results <- tibble::tibble()

  for (
    factor_name in
    factor_names
  ) {
    factor_data <-
      factor_scores %>%
      dplyr::filter(
        Factor == factor_name
      ) %>%
      dplyr::select(
        Factor_value,
        Fibrosis_percent,
        Age_years
      ) %>%
      tidyr::drop_na()

    if (
      nrow(factor_data) < 6L ||
      stats::sd(
        factor_data$Fibrosis_percent
      ) == 0
    ) {
      next
    }

    factor_data <- factor_data %>%
      dplyr::mutate(
        fibrosis_scaled =
          as.numeric(
            scale(
              Fibrosis_percent
            )
          ),
        age_scaled =
          as.numeric(
            scale(Age_years)
          )
      )

    fibrosis_fit <- lm(
      Factor_value ~
        fibrosis_scaled +
        age_scaled,
      data = factor_data
    )

    fibrosis_coefficient <-
      broom::tidy(
        fibrosis_fit,
        conf.int = TRUE
      ) %>%
      dplyr::filter(
        term ==
          "fibrosis_scaled"
      )

    results <- dplyr::bind_rows(
      results,
      tibble::tibble(
        Factor = factor_name,
        n = nrow(factor_data),
        beta =
          fibrosis_coefficient$
            estimate,
        standard_error =
          fibrosis_coefficient$
            std.error,
        confidence_interval_low =
          fibrosis_coefficient$
            conf.low,
        confidence_interval_high =
          fibrosis_coefficient$
            conf.high,
        t_statistic =
          fibrosis_coefficient$
            statistic,
        p_value =
          fibrosis_coefficient$
            p.value,
        model_R2 =
          summary(
            fibrosis_fit
          )$r.squared,
        adjusted_model_R2 =
          summary(
            fibrosis_fit
          )$adj.r.squared
      )
    )
  }

  if (nrow(results) == 0L) {
    return(results)
  }

  results %>%
    dplyr::mutate(
      FDR = p.adjust(
        p_value,
        method = "BH"
      )
    ) %>%
    dplyr::arrange(FDR)
}

Tune the number of proteomics factors

Three seeds are fitted for every candidate number of factors. Each complete execution refits all tuning models and overwrites the corresponding tuning files.

proteomics_tuning_results <-
  tibble::tibble()

for (n_factors in candidate_factors) {
  for (tuning_seed in tuning_seeds) {
    tuning_model_file <- file.path(
      proteomics_output_dir,
      paste0(
        "MOFA_proteomics_tuning_K",
        sprintf("%02d", n_factors),
        "_seed",
        tuning_seed,
        ".hdf5"
      )
    )

    if (file.exists(tuning_model_file)) {
      unlink(tuning_model_file)
    }

    message(
      "Starting proteomics model: K = ",
      n_factors,
      ", seed = ",
      tuning_seed
    )

    tuning_object <- MOFA2::create_mofa(
      proteomics_data
    )

    tuning_data_options <-
      MOFA2::get_default_data_options(
        tuning_object
      )

    tuning_data_options$scale_views <- TRUE

    tuning_model_options <-
      MOFA2::get_default_model_options(
        tuning_object
      )

    tuning_model_options$num_factors <-
      n_factors

    tuning_model_options$likelihoods[] <-
      "gaussian"

    tuning_training_options <-
      MOFA2::get_default_training_options(
        tuning_object
      )

    tuning_training_options$maxiter <-
      tuning_maxiter

    tuning_training_options$convergence_mode <-
      mofa_convergence_mode

    tuning_training_options$freqELBO <-
      mofa_freq_elbo

    tuning_training_options$drop_factor_threshold <-
      -1

    tuning_training_options$seed <-
      tuning_seed

    tuning_training_options$verbose <-
      FALSE

    tuning_object <- MOFA2::prepare_mofa(
      object = tuning_object,
      data_options = tuning_data_options,
      model_options = tuning_model_options,
      training_options =
        tuning_training_options
    )

    MOFA2::run_mofa(
      tuning_object,
      outfile = tuning_model_file,
      use_basilisk = TRUE
    )

    rm(tuning_object)
    gc()

    tuning_model <- MOFA2::load_model(
      tuning_model_file,
      sort_factors = TRUE,
      remove_inactive_factors = FALSE
    )

    tuning_r2 <-
      MOFA2::get_variance_explained(
        tuning_model
      )$r2_per_factor[[1]]

    weakest_factor <-
      tail(
        rownames(tuning_r2),
        1
      )

    weakest_factor_r2 <-
      tuning_r2[
        weakest_factor,
        ,
        drop = TRUE
      ]

    new_tuning_result <- tibble::tibble(
      n_factors = n_factors,
      seed = tuning_seed,
      final_ELBO =
        MOFA2::get_elbo(tuning_model),
      weakest_factor = weakest_factor,
      weakest_factor_total_R2 =
        sum(
          weakest_factor_r2,
          na.rm = TRUE
        ),
      weakest_factor_Olink_R2 =
        unname(
          weakest_factor_r2["Olink"]
        ),
      weakest_factor_LCMS_R2 =
        unname(
          weakest_factor_r2["LCMS"]
        ),
      n_factors_total_R2_at_least_cutoff =
        sum(
          rowSums(
            tuning_r2,
            na.rm = TRUE
          ) >= last_factor_r2_cutoff
        ),
      model_file = tuning_model_file
    )

    proteomics_tuning_results <-
      dplyr::bind_rows(
        proteomics_tuning_results,
        new_tuning_result
      ) %>%
      dplyr::arrange(
        n_factors,
        seed
      )

    write.csv(
      proteomics_tuning_results,
      proteomics_tuning_summary_file,
      row.names = FALSE
    )

    message(
      "Finished proteomics model: K = ",
      n_factors,
      ", seed = ",
      tuning_seed
    )

    rm(tuning_model)
    gc()
  }

wrap_gsea_labels <- function(
    x,
    width = 41
) {
  vapply(
    x,
    function(label) {
      label <-
        stringr::str_squish(
          as.character(label)
        )

      if (
        nchar(label) <= width
      ) {
        return(label)
      }

      words <-
        stringr::str_split(
          label,
          "\\s+"
        )[[1]]

      lines <- c(
        "",
        "",
        ""
      )

      current_line <- 1L

      for (
        word in words
      ) {
        candidate <-
          if (
            lines[current_line] == ""
          ) {
            word
          } else {
            paste(
              lines[current_line],
              word
            )
          }

        if (
          nchar(candidate) <= width
        ) {
          lines[current_line] <-
            candidate
        } else if (
          current_line < 3L
        ) {
          current_line <-
            current_line + 1L

          lines[current_line] <-
            word
        } else {
          lines[current_line] <-
            paste0(
              substr(
                lines[current_line],
                1,
                width - 1
              ),
              "…"
            )

          break
        }
      }

      if (
        nchar(lines[3]) > width
      ) {
        lines[3] <-
          paste0(
            substr(
              lines[3],
              1,
              width - 1
            ),
            "…"
          )
      }

      paste(
        lines[
          lines != ""
        ],
        collapse = "\n"
      )
    },
    FUN.VALUE = character(1)
  )
}

extract_factor1_gsea_heatmap_matrix <- function(
    feature_table,
    data_views,
    sample_order
) {
  extracted_rows <- vector(
    mode = "list",
    length = nrow(feature_table)
  )

  row_found <- logical(
    nrow(feature_table)
  )

  resolved_features <- character(
    nrow(feature_table)
  )

  for (
    feature_index in
    seq_len(nrow(feature_table))
  ) {
    current_view <-
      as.character(
        feature_table$View[
          feature_index
        ]
      )

    current_feature <-
      as.character(
        feature_table$Feature[
          feature_index
        ]
      )

    if (
      !current_view %in%
        names(data_views)
    ) {
      next
    }

    feature_candidates <- unique(
      c(
        current_feature,

        paste0(
          current_feature,
          "_",
          current_view
        ),

        paste0(
          current_feature,
          "_",
          dplyr::case_when(
            current_view == "Olink" ~
              "OLINK",

            current_view == "LCMS" ~
              "MS",

            TRUE ~
              current_view
          )
        )
      )
    )

    resolved_feature <-
      feature_candidates[
        feature_candidates %in%
          rownames(
            data_views[[
              current_view
            ]]
          )
      ][1]

    if (
      length(resolved_feature) == 0L ||
      is.na(resolved_feature)
    ) {
      next
    }

    extracted_rows[[
      feature_index
    ]] <-
      as.numeric(
        data_views[[
          current_view
        ]][
          resolved_feature,
          sample_order,
          drop = TRUE
        ]
      )

    resolved_features[
      feature_index
    ] <- resolved_feature

    row_found[
      feature_index
    ] <- TRUE
  }

  if (!any(row_found)) {
    stop(
      "None of the selected features were found in the proteomics matrices."
    )
  }

  if (any(!row_found)) {
    warning(
      "The following selected features were not found and were omitted: ",
      paste(
        paste0(
          feature_table$Feature[
            !row_found
          ],
          " (",
          feature_table$View[
            !row_found
          ],
          ")"
        ),
        collapse = ", "
      )
    )
  }

  retained_feature_table <-
    feature_table[
      row_found,
      ,
      drop = FALSE
    ] %>%
    dplyr::mutate(
      Resolved_feature =
        resolved_features[
          row_found
        ],

      Display_feature =
        Feature,

      Modality =
        dplyr::case_when(
          View == "Olink" ~
            "OLINK",

          View == "LCMS" ~
            "MS",

          TRUE ~
            as.character(View)
        ),

      Internal_row_id =
        make.unique(
          paste(
            Pathway,
            View,
            Feature,
            sep = "__"
          )
        )
    )

  expression_matrix <-
    do.call(
      rbind,
      extracted_rows[
        row_found
      ]
    )

  colnames(expression_matrix) <-
    sample_order

  rownames(expression_matrix) <-
    retained_feature_table$
      Internal_row_id

  list(
    matrix =
      expression_matrix,

    feature_table =
      retained_feature_table
  )
}


prepare_factor1_gsea_heatmap_z_scores <- function(
    expression_matrix
) {
  imputed_matrix <-
    expression_matrix

  for (
    row_index in
    seq_len(nrow(imputed_matrix))
  ) {
    row_values <-
      imputed_matrix[
        row_index,
        ,
        drop = TRUE
      ]

    finite_values <-
      row_values[
        is.finite(row_values)
      ]

    if (
      length(finite_values) == 0L
    ) {
      next
    }

    row_values[
      !is.finite(row_values)
    ] <-
      stats::median(
        finite_values
      )

    imputed_matrix[
      row_index,
    ] <- row_values
  }

  row_means <-
    rowMeans(
      imputed_matrix,
      na.rm = TRUE
    )

  row_standard_deviations <-
    apply(
      imputed_matrix,
      1,
      stats::sd,
      na.rm = TRUE
    )

  row_standard_deviations[
    !is.finite(
      row_standard_deviations
    ) |
      row_standard_deviations == 0
  ] <- 1

  z_score_matrix <-
    sweep(
      imputed_matrix,
      1,
      row_means,
      "-"
    )

  z_score_matrix <-
    sweep(
      z_score_matrix,
      1,
      row_standard_deviations,
      "/"
    )

  pmax(
    pmin(
      z_score_matrix,
      2
    ),
    -2
  )
}


make_grey_annotation_scale <- function(
    values,
    dark_at = c(
      "low",
      "high"
    )
) {
  dark_at <-
    match.arg(
      dark_at
    )

  finite_values <-
    values[
      is.finite(values)
    ]

  value_range <-
    if (
      length(finite_values) == 0L
    ) {
      c(0, 1)
    } else {
      range(
        finite_values,
        na.rm = TRUE
      )
    }

  if (
    value_range[1] ==
      value_range[2]
  ) {
    value_range <-
      value_range +
      c(
        -0.5,
        0.5
      )
  }

  annotation_colours <-
    if (
      dark_at == "low"
    ) {
      c(
        "grey10",
        "grey95"
      )
    } else {
      c(
        "grey95",
        "grey10"
      )
    }

  circlize::colorRamp2(
    value_range,
    annotation_colours
  )
}

        MOFA2::get_elbo(tuning_model),
      weakest_factor = weakest_factor,
      weakest_factor_total_R2 =
        sum(
          weakest_factor_r2,
          na.rm = TRUE
        ),
      weakest_factor_RNA_R2 =
        unname(
          weakest_factor_r2[
            "RNA_pseudobulk"
          ]
        ),
      weakest_factor_Olink_R2 =
        unname(
          weakest_factor_r2[
            "Olink"
          ]
        ),
      weakest_factor_LCMS_R2 =
        unname(
          weakest_factor_r2[
            "LCMS"
          ]
        ),
      n_factors_total_R2_at_least_cutoff =
        sum(
          rowSums(
            tuning_r2,
            na.rm = TRUE
          ) >= last_factor_r2_cutoff
        ),
      model_file = tuning_model_file
    )

    matched_tuning_results <-
      dplyr::bind_rows(
        matched_tuning_results,
        new_tuning_result
      ) %>%
      dplyr::arrange(
        n_factors,
        seed
      )

    write.csv(
      matched_tuning_results,
      matched_tuning_summary_file,
      row.names = FALSE
    )

    message(
      "Finished matched model: K = ",
      n_factors,
      ", seed = ",
      tuning_seed
    )

    rm(tuning_model)
    gc()
  }
}

for (

      RV_group = factor(
        as.character(RV_group),

          rv_group_comparison_levels
      )
    ) %>%
    droplevels()

  if (
    nlevels(factor_data$RV_group) < 2L
  ) {
    next
  }

  group_counts <-
    table(
      factor_data$RV_group
    )

  group_fit <- lm(
    Factor_value ~
      RV_group +
      scale(Age_years),
    data = factor_data
  )

  group_test <- drop1(
    group_fit,
    test = "F"
  )

  matched_group_global_results <-
    dplyr::bind_rows(
      matched_group_global_results,
      tibble::tibble(
        Factor = factor_name,
        comparison =
          "Compensated versus Decompensated",
        n = nrow(factor_data),
        n_compensated =
          unname(
            group_counts["Compensated"]
          ),
        n_decompensated =
          unname(
            group_counts["Decompensated"]
          ),
        F_statistic =
          group_test[
            "RV_group",
            "F value"
          ],
        p_value =
          group_test[
            "RV_group",
            "Pr(>F)"
          ],
        model_R2 =
          summary(group_fit)$r.squared,
        adjusted_model_R2 =
          summary(group_fit)$adj.r.squared
      )
    )

  factor_emmeans <- emmeans::emmeans(
    group_fit,
    ~ RV_group
  )

  factor_contrasts <-
    emmeans::contrast(
      factor_emmeans,
      method = list(
        "Decompensated - Compensated" =
          c(-1, 1)
      ),
      adjust = "none"
    ) %>%
    summary(
      infer = c(TRUE, TRUE)
    ) %>%
    as.data.frame() %>%
    dplyr::mutate(
      Factor = factor_name,
      n = nrow(factor_data),
      n_compensated =
        unname(

      n_decompensated
    )

      matched_group_pairwise_results,
      factor_contrasts
    )
}

extract_matched_factor2_heatmap_matrix <- function(
    feature_table,
    data_views,
    sample_order
) {
  extracted_rows <- vector(
    mode = "list",
    length = nrow(
      feature_table
    )
  )

  row_found <- logical(
    nrow(
      feature_table
    )
  )

  resolved_features <- character(
    nrow(
      feature_table
    )
  )

  for (
    feature_index in
    seq_len(
      nrow(
        feature_table
      )
    )
  ) {
    current_view <-
      as.character(
        feature_table$View[
          feature_index
        ]
      )

    current_feature <-
      as.character(
        feature_table$Feature_actual[
          feature_index
        ]
      )

    if (
      !current_view %in%
        names(
          data_views
        )
    ) {
      next
    }

    feature_candidates <- unique(
      c(
        current_feature,

        as.character(
          feature_table$Feature[
            feature_index
          ]
        ),

        paste0(
          as.character(
            feature_table$Feature[
              feature_index
            ]
          ),
          "_",
          current_view
        ),

        paste0(
          as.character(
            feature_table$Feature[
              feature_index
            ]
          ),
          "_",
          dplyr::case_when(
            current_view == "Olink" ~
              "OLINK",

            current_view == "LCMS" ~
              "MS",

            current_view == "RNA_pseudobulk" ~
              "RNA_pseudobulk",

            TRUE ~
              current_view
          )
        )
      )
    )

    resolved_feature <-
      feature_candidates[
        feature_candidates %in%
          rownames(
            data_views[[
              current_view
            ]]
          )
      ][1]

    if (
      length(
        resolved_feature
      ) == 0L ||
      is.na(
        resolved_feature
      )
    ) {
      next
    }

    extracted_rows[[
      feature_index
    ]] <-
      as.numeric(
        data_views[[
          current_view
        ]][
          resolved_feature,
          sample_order,
          drop = TRUE
        ]
      )

    resolved_features[
      feature_index
    ] <-
      resolved_feature

    row_found[
      feature_index
    ] <-
      TRUE
  }


  if (
    !any(
      row_found
    )
  ) {
    stop(
      "None of the selected matched-model features were found in matched_data."
    )
  }


  if (
    any(
      !row_found
    )
  ) {
    warning(
      "The following selected features were not found and were omitted: ",
      paste(
        paste0(
          feature_table$Feature[
            !row_found
          ],
          " (",
          feature_table$View[
            !row_found
          ],
          ")"
        ),
        collapse = ", "
      )
    )
  }


  retained_feature_table <-
    feature_table[
      row_found,
      ,
      drop = FALSE
    ] %>%
    dplyr::mutate(
      Resolved_feature =
        resolved_features[
          row_found
        ],

      Display_feature =
        Feature,

      Internal_row_id =
        make.unique(
          paste(
            Annotation,
            View,
            Feature,
            sep = "__"
          )
        )
    )


  expression_matrix <-
    do.call(
      rbind,
      extracted_rows[
        row_found
      ]
    )


  colnames(
    expression_matrix
  ) <-
    sample_order


  rownames(
    expression_matrix
  ) <-
    retained_feature_table$
      Internal_row_id


  list(
    matrix =
      expression_matrix,

    feature_table =
      retained_feature_table
  )
}


prepare_matched_heatmap_z_scores <- function(
    expression_matrix
) {
  imputed_matrix <-
    expression_matrix


  for (
    row_index in
    seq_len(
      nrow(
        imputed_matrix
      )
    )
  ) {
    row_values <-
      imputed_matrix[
        row_index,
        ,
        drop = TRUE
      ]

    finite_values <-
      row_values[
        is.finite(
          row_values
        )
      ]

    if (
      length(
        finite_values
      ) == 0L
    ) {
      next
    }

    row_values[
      !is.finite(
        row_values
      )
    ] <-
      stats::median(
        finite_values
      )

    imputed_matrix[
      row_index,
    ] <-
      row_values
  }


  row_means <-
    rowMeans(
      imputed_matrix,
      na.rm = TRUE
    )


  row_standard_deviations <-
    apply(
      imputed_matrix,
      1,
      stats::sd,
      na.rm = TRUE
    )


  row_standard_deviations[
    !is.finite(
      row_standard_deviations
    ) |
      row_standard_deviations == 0
  ] <- 1


  z_score_matrix <-
    sweep(
      imputed_matrix,
      1,
      row_means,
      "-"
    )


  z_score_matrix <-
    sweep(
      z_score_matrix,
      1,
      row_standard_deviations,
      "/"
    )


  pmax(
    pmin(
      z_score_matrix,
      2
    ),
    -2
  )
}

  )

wrap_factor2_gsea_labels <- function(
    x,
    width = 41
) {
  vapply(
    x,
    function(label) {
      label <-
        stringr::str_squish(
          as.character(label)
        )

      if (
        nchar(label) <= width
      ) {
        return(label)
      }

      words <-
        stringr::str_split(
          label,
          "\\s+"
        )[[1]]

      lines <- c(
        "",
        "",
        ""
      )

      current_line <- 1L

      for (
        word in words
      ) {
        candidate <-
          if (
            lines[current_line] == ""
          ) {
            word
          } else {
            paste(
              lines[current_line],
              word
            )
          }

        if (
          nchar(candidate) <= width
        ) {
          lines[current_line] <-
            candidate
        } else if (
          current_line < 3L
        ) {
          current_line <-
            current_line + 1L

          lines[current_line] <-
            word
        } else {
          lines[current_line] <-
            paste0(
              substr(
                lines[current_line],
                1,
                width - 1
              ),
              "…"
            )

          break
        }
      }

      if (
        nchar(lines[3]) > width
      ) {
        lines[3] <-
          paste0(
            substr(
              lines[3],
              1,
              width - 1
            ),
            "…"
          )
      }

      paste(
        lines[
          lines != ""
        ],
        collapse = "\n"
      )
    },
    FUN.VALUE = character(1)
  )
}


make_factor2_simplified_gsea_plot <- function(
    gsea_object,
    layer_label
) {
  if (
    is.null(gsea_object) ||
    nrow(
      as.data.frame(
        gsea_object
      )
    ) == 0L
  ) {
    return(
      ggplot() +
        annotate(
          "text",
          x = 1,
          y = 1,
          label =
            "No GSEA results"
        ) +
        labs(
          title =
            paste0(
              layer_label,
              " | GSEA | Gene Ontology"
            )
        ) +
        theme_void()
    )
  }

  gsea_results <-
    as.data.frame(
      gsea_object
    ) %>%
    tibble::as_tibble() %>%
    dplyr::filter(
      is.finite(NES),
      is.finite(p.adjust),
      p.adjust <
        factor2_gsea_fdr_cutoff
    )

  if (
    nrow(gsea_results) == 0L
  ) {
    return(
      ggplot() +
        annotate(
          "text",
          x = 1,
          y = 1,
          label =
            paste0(
              "No terms with FDR < ",
              factor2_gsea_fdr_cutoff
            )
        ) +
        labs(
          title =
            paste0(
              layer_label,
              " | GSEA | Gene Ontology"
            )
        ) +
        theme_void()
    )
  }

  positive_terms <-
    gsea_results %>%
    dplyr::filter(
      NES > 0
    ) %>%
    dplyr::arrange(
      dplyr::desc(NES)
    ) %>%
    dplyr::slice_head(
      n =
        factor2_gsea_top_positive
    )

  negative_terms <-
    gsea_results %>%
    dplyr::filter(
      NES < 0
    ) %>%
    dplyr::arrange(NES) %>%
    dplyr::slice_head(
      n =
        factor2_gsea_top_negative
    )

  plot_data <-
    dplyr::bind_rows(
      positive_terms,
      negative_terms
    ) %>%
    dplyr::mutate(
      Description_clean =
        stringr::str_remove(
          Description,
          "^\\[(BP|CC|MF)\\]\\s*"
        )
    )

  if (
    "pvalue" %in%
      colnames(plot_data)
  ) {
    plot_data <-
      plot_data %>%
      dplyr::mutate(
        p_for_colour =
          dplyr::if_else(
            is.finite(pvalue) &
              pvalue > 0,
            pvalue,
            p.adjust
          )
      )
  } else {
    plot_data <-
      plot_data %>%
      dplyr::mutate(
        p_for_colour =
          p.adjust
      )
  }

  plot_data <-
    plot_data %>%
    dplyr::mutate(
      colour_score =
        -log10(
          pmax(
            p_for_colour,
            .Machine$double.xmin
          )
        ) *
        sign(NES),

      Description_wrapped =
        wrap_factor2_gsea_labels(
          Description_clean,
          width = 41
        )
    ) %>%
    dplyr::arrange(NES) %>%
    dplyr::mutate(
      Description_wrapped =
        factor(
          Description_wrapped,
          levels =
            unique(
              Description_wrapped
            )
        )
    )

  colour_limit <-
    unname(
      stats::quantile(
        abs(
          plot_data$
            colour_score
        ),
        probs = 0.80,
        na.rm = TRUE
      )
    )

  if (
    length(colour_limit) != 1L ||
    !is.finite(colour_limit) ||
    colour_limit == 0
  ) {
    colour_limit <-
      max(
        abs(
          plot_data$
            colour_score
        ),
        na.rm = TRUE
      )
  }

  ggplot(
    plot_data,
    aes(
      x =
        Description_wrapped,
      y = NES,
      fill =
        colour_score
    )
  ) +
    geom_col(
      width = 0.9
    ) +
    coord_flip() +
    scale_fill_gradient2(
      low = "#004C8C",
      mid = "white",
      high = "#A63A2A",
      midpoint = 0,
      limits = c(
        -colour_limit,
        colour_limit
      ),
      oob =
        scales::squish,
      name =
        expression(
          -log[10](p) %*% sign(NES)
        )
    ) +
    labs(
      title =
        paste0(
          layer_label,
          " | GSEA | Gene Ontology"
        ),
      x = NULL,
      y =
        "Normalized enrichment score (NES)"
    ) +
    theme_bw(
      base_size = 10
    ) +
    theme(
      text =
        element_text(
          color = "black"
        ),

      plot.title =
        element_text(
          hjust = 0.5,
          face = "bold",
          size = 11
        ),

      axis.text =
        element_text(
          color = "black"
        ),

      axis.title.x =
        element_text(
          color = "black",
          size = 12
        ),

      axis.text.y =
        element_text(
          size = 9,
          color = "black",
          lineheight = 0.7
        ),

      panel.grid.minor =
        element_blank(),

      legend.position =
        "bottom",

      legend.direction =
        "horizontal",

      legend.title =
        element_text(
          size = 9
        ),

      legend.text =
        element_text(
          size = 9
        )
    ) +
    guides(
      fill =
        guide_colorbar(
          barwidth =
            grid::unit(
              6,
              "cm"
            ),

          barheight =
            grid::unit(
              0.3,
              "cm"
            ),

          title.position =
            "top",

          title.hjust =
            0.5
        )
    )
}
