group_design <- function(metadata, samples) {
    stopifnot(!anyDuplicated(metadata$Sample), !anyDuplicated(samples), all(samples %in% metadata$Sample))
    d <- metadata[match(samples, metadata$Sample), , drop = FALSE]
    stopifnot(identical(as.character(d$Sample), as.character(samples)), all(d$Group %in% c("HC", "DCM")))
    d$Group <- factor(d$Group, levels = c("HC", "DCM"))
    stopifnot(all(table(d$Group) >= 3), all(is.finite(d$Age_years)))
    design <- model.matrix(~Group + Age_years, data = d)
    stopifnot(qr(design)$rank == ncol(design), nrow(design) > ncol(design))
    rownames(design) <- d$Sample
    list(metadata = d, design = design)
}
fit_group <- function(counts, metadata) {
    stopifnot(all(counts >= 0), all(counts == round(counts)))
    keep_samples <- colSums(counts) > 50000
    s <- colnames(counts)[keep_samples]
    s <- s[s %in% metadata$Sample[is.finite(metadata$Age_years)]]
    a <- group_design(metadata, s)
    y <- edgeR::DGEList(counts[, s, drop = FALSE])
    keep <- edgeR::filterByExpr(y)
    stopifnot(any(keep))
    y <- y[keep, , keep.lib.sizes = FALSE]
    y <- edgeR::normLibSizes(y)
    y <- edgeR::estimateDisp(y, a$design)
    fit <- edgeR::glmQLFit(y, a$design, robust = TRUE)
    test <- edgeR::glmQLFTest(fit, coef = "GroupDCM")
    z <- edgeR::topTags(test, n = Inf)$table
    z$gene <- rownames(z)
    z <- z[, c("gene", setdiff(names(z), "gene"))]
    list(results = z, design = a$design, metadata = a$metadata, pseudobulk = y, excluded_samples = setdiff(colnames(counts), 
        s))
}

objects <- list()
simplified <- list()
ranked_genes <- list()
for (ct in de_summary$cell_type[de_summary$status == "completed"]) {
    id <- gsub(" ", "_", ct, fixed = TRUE)
    de <- read.csv(file.path(O, paste0("A_", id, "_DE_DCM_vs_HC.csv")))
    de$metric <- de$logFC * -log10(pmax(de$PValue, .Machine$double.xmin))
    mapping <- bitr(de$gene, fromType = "SYMBOL", toType = "ENTREZID", OrgDb = org.Hs.eg.db)
    ranked <- merge(de, mapping, by.x = "gene", by.y = "SYMBOL", sort = FALSE)
    ranked <- ranked[order(-abs(ranked$metric), ranked$gene, ranked$ENTREZID), ]
    ranked <- ranked[!duplicated(ranked$ENTREZID), ]
    ranked <- ranked[order(-ranked$metric, ranked$ENTREZID), ]
    ranks <- setNames(ranked$metric, ranked$ENTREZID)
    ranked_genes[[ct]] <- ranked
    checkpoint <- here::here("data", "processed", "DCM_control_GSEA", paste0("A_", id, "_GSEA_GO.rds"))
    set.seed(20240606)
    if (file.exists(checkpoint)) {
        go <- readRDS(checkpoint)
        stopifnot(isTRUE(all.equal(go@geneList, ranks, tolerance = 1e-12)))
    }
    else {
        go <- gseGO(geneList = ranks, OrgDb = org.Hs.eg.db, ont = "ALL", keyType = "ENTREZID", minGSSize = 10, 
            nPermSimple = if (ct == "Cardiomyocytes") 
                100000L
            else 1000000L, eps = if (ct == "Cardiomyocytes") 
                1e-10
            else 0, pvalueCutoff = 1, pAdjustMethod = "BH", BPPARAM = BiocParallel::SerialParam(), verbose = FALSE)
    }
    objects[[ct]] <- go
    significant <- go
    significant@result <- go@result %>% filter(is.finite(p.adjust), p.adjust < 0.05, is.finite(NES))
    if (nrow(significant@result) == 0) 
        next
    simplified[[ct]] <- if (nrow(significant@result) == 1) 
        significant
    else clusterProfiler::simplify(significant, cutoff = 0.6, by = "p.adjust", select_fun = min, measure = "Wang")
}
