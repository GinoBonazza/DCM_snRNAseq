# Resolve each requested lncRNA against the actual feature names of a dataset.
# The resolver tries, in order: gene symbol, exact Ensembl ID, then a versioned
# Ensembl ID (e.g. ENSG00000130600.1). Missing genes are retained as NA so that
# combined plots keep the same seven candidate columns across datasets.
resolve_lncRNA_features <- function(object) {
  object_features <- rownames(object)

  matched <- vapply(
    names(lncrna_symbol),
    FUN = function(gene_name) {
      symbol <- unname(lncrna_symbol[gene_name])
      ensembl <- unname(lncrna_ens[gene_name])

      if (!is.na(symbol) && symbol %in% object_features) {
        return(symbol)
      }

      if (!is.na(ensembl) && ensembl %in% object_features) {
        return(ensembl)
      }

      if (!is.na(ensembl)) {
        versioned_match <- grep(
          paste0("^", ensembl, "\\."),
          object_features,
          value = TRUE
        )
        if (length(versioned_match) > 0) {
          return(versioned_match[1])
        }
      }

      return(NA_character_)
    },
    FUN.VALUE = character(1)
  )

  matched
}

check_lncRNA_features <- function(object) {
  matched <- resolve_lncRNA_features(object)

  data.frame(
    lncRNA = names(lncrna_symbol),
    symbol = unname(lncrna_symbol),
    ensembl = unname(lncrna_ens),
    matched_feature = unname(matched),
    available = !is.na(matched),
    stringsAsFactors = FALSE
  )
}

  available_named <- get_available_features(object, features)
  available_features <- unname(available_named)
  feature_to_label <- setNames(names(available_named), unname(available_named))

  if (length(available_features) > 0) {
    tmp <- DotPlot(
      object,
      features = available_features,
      group.by = "cell_type_simple"
    )$data %>%
      as.data.frame() %>%
      dplyr::transmute(
        dataset = dataset_name,
        cell_type = as.character(id),
        feature = as.character(features.plot),
        gene = unname(feature_to_label[as.character(features.plot)]),
        pct.exp = pct.exp,
        avg.exp.scaled = avg.exp.scaled
      )
  } else {
    tmp <- data.frame(
      dataset = character(),
      cell_type = character(),
      feature = character(),
      gene = character(),
      pct.exp = numeric(),
      avg.exp.scaled = numeric()
    )
  }

  availability <- data.frame(
    gene = names(features),
    feature_available = !is.na(unname(features)) & unname(features) %in% rownames(object),
    stringsAsFactors = FALSE
  )

  grid <- tidyr::expand_grid(
    dataset = dataset_name,
    cell_type = cell_type_order,
    gene = names(features)
  )

  tmp <- grid %>%
    dplyr::left_join(availability, by = "gene") %>%
    dplyr::left_join(
      tmp %>% dplyr::select(dataset, cell_type, gene, pct.exp, avg.exp.scaled),
      by = c("dataset", "cell_type", "gene")
    )

  tmp$cell_type <- factor(tmp$cell_type, levels = rev(cell_type_order))
  tmp$gene <- factor(tmp$gene, levels = names(features))
  tmp
}

# Individual datasets: use Seurat-like default scales.
# The percent-expressed scale adapts to each dataset independently.
# Positions with exactly 0% expression are retained in the grid but no dot is drawn.
plot_dotplot_data <- function(data, title = NULL) {
  gene_levels <- levels(data$gene)

  ggplot(data, aes(x = gene, y = cell_type)) +
    geom_point(
      data = data %>% dplyr::filter(!is.na(pct.exp) & pct.exp > 0),
      aes(size = pct.exp, color = avg.exp.scaled)
    ) +
    scale_radius(
      name = "Percent expressed",
      range = c(0, 6)
    ) +
    scale_color_gradient(
      name = "Average expression",
      low = "lightgrey",
      high = "blue"
    ) +
    scale_x_discrete(drop = FALSE, limits = gene_levels) +
    scale_y_discrete(drop = FALSE) +
    labs(
      x = NULL,
      y = NULL,
      title = title
    ) +
    theme_classic() +
    theme(
      axis.text.x = element_text(angle = 45, hjust = 1, size = 11),
      axis.text.y = element_text(size = 11),
      plot.title = element_text(hjust = 0.5, face = "bold")
    )
}

# Combined datasets: one shared legend and fixed scales across all datasets.
# max_pct only sets the common legend/size scale; the underlying percentages are unchanged.
plot_combined_dotplot <- function(data, dataset_order, title = NULL,
                                  max_pct = 100, pct_breaks = NULL) {
  data$dataset <- factor(data$dataset, levels = dataset_order)
  gene_levels <- levels(data$gene)

  if (is.null(pct_breaks)) {
    pct_breaks <- pretty(c(0, max_pct), n = 4)
    pct_breaks <- pct_breaks[pct_breaks > 0 & pct_breaks <= max_pct]
  }

  ggplot(data, aes(x = gene, y = cell_type)) +
    geom_point(
      data = data %>% dplyr::filter(!is.na(pct.exp) & pct.exp > 0),
      aes(size = pct.exp, color = avg.exp.scaled)

    facet_grid(. ~ dataset, scales = "free_x", space = "free_x") +
    scale_radius(
      name = "Percent expressed",
      range = c(0, 6),
      limits = c(0, max_pct),
      breaks = pct_breaks
    ) +
    scale_color_gradient(
      name = "Average expression",
      low = "lightgrey",
      high = "blue",
      limits = c(-2.5, 2.5),
      oob = scales::squish
    ) +
    scale_x_discrete(drop = FALSE, limits = gene_levels) +
    scale_y_discrete(drop = FALSE) +
    labs(
      x = NULL,
      y = NULL,
      title = title
    ) +
    theme_classic() +
    theme(
      axis.text.x = element_text(angle = 45, hjust = 1, size = 10),
      axis.text.y = element_text(size = 11),
      strip.text = element_text(face = "bold", size = 11),
      strip.background = element_blank(),
      panel.spacing.x = grid::unit(1.2, "lines"),
      plot.title = element_text(hjust = 0.5, face = "bold")
    )
}

make_stacked_vlnplot <- function(object, features, dataset_title, alpha = 0.3) {
  available_named <- get_available_features(object, features)
  available_features <- unname(available_named)
  available_labels <- names(available_named)

  if (length(available_features) == 0) {
    stop(paste0("None of the requested features are present in ", dataset_title, "."))
  }

  Idents(object) <- "cell_type_simple"
  p <- Stacked_VlnPlot(
    object,
    features = available_features,
    x_lab_rotate = TRUE,
    pt.size = 0.001,
    plot_spacing = 0.05
  )

  for (i in seq_along(p)) {
    if (length(p[[i]][[1]]$layers) >= 2) {
      p[[i]][[1]]$layers[[2]]$aes_params$alpha <- alpha
    }
    p[[i]][[1]] <- p[[i]][[1]] + labs(y = available_labels[i])
  }

  wrap_plots(p) +
    plot_annotation(title = dataset_title) &
    theme(plot.title = element_text(hjust = 0.5, face = "bold", size = 15))
}

# Feature plots use the same scCustomize palette as the original presentation.
# Genes absent from a released dataset are still shown as a grey-only UMAP panel.
make_featureplot <- function(object, features, dataset_title) {
  available_named <- get_available_features(object, features)
  available_labels <- names(available_named)
  requested_labels <- names(features)

  plot_list <- vector("list", length(requested_labels))
  names(plot_list) <- requested_labels

  if (length(available_named) > 0) {
    available_plots <- FeaturePlot_scCustom(
      seurat_object = object,
      features = unname(available_named),
      colors_use = viridis_plasma_dark_high,
      na_color = "lightgray",
      order = TRUE,
      combine = FALSE
    )

    for (i in seq_along(available_plots)) {
      label_here <- available_labels[i]
      available_plots[[i]] <- available_plots[[i]] + ggtitle(label_here)
      plot_list[[label_here]] <- available_plots[[i]]
    }
  }

  missing_labels <- setdiff(requested_labels, available_labels)

  if (length(missing_labels) > 0) {
    for (label_here in missing_labels) {
      object$.feature_not_available <- "not available"
      plot_list[[label_here]] <- DimPlot(
        object,
        group.by = ".feature_not_available",
        cols = "lightgray"
      ) +
        NoLegend() +
        ggtitle(label_here)
    }
  }

  wrap_plots(plot_list, ncol = 4) +
    plot_annotation(title = dataset_title) &
    theme(plot.title = element_text(hjust = 0.5, face = "bold", size = 16))
}

# Make sample-level pseudobulk expression data from the unfiltered DGEList.
# A candidate is plotted if it has at least one non-zero raw pseudobulk count.
# Thus genes detected but excluded by filterByExpr remain visible, whereas genes
# with zero counts in every sample are not shown.
make_pseudobulk_box_data <- function(pseudocounts_all, metadata, sample_col, group_col, features) {
  raw_counts <- pseudocounts_all$counts
  detected_features <- rownames(raw_counts)[rowSums(raw_counts) > 0]

  pseudocounts_plot <- edgeR::normLibSizes(pseudocounts_all)
  logcpm <- edgeR::cpm(pseudocounts_plot, log = TRUE, prior.count = 1)

  available_named <- features[
    !is.na(unname(features)) &
      unname(features) %in% rownames(logcpm) &
      unname(features) %in% detected_features
  ]

  if (length(available_named) == 0) {
    stop("None of the requested features have non-zero pseudobulk counts.")
  }

  feature_to_label <- setNames(names(available_named), unname(available_named))

  expr <- as.data.frame(
    t(logcpm[unname(available_named), , drop = FALSE])
  ) %>%
    tibble::rownames_to_column("sample") %>%
    tidyr::pivot_longer(
      cols = -sample,
      names_to = "feature",
      values_to = "logCPM"
    ) %>%
    dplyr::mutate(
      gene = unname(feature_to_label[feature])
    )

  meta <- data.frame(
    sample = as.character(metadata[[sample_col]]),
    group = as.character(metadata[[group_col]]),
    stringsAsFactors = FALSE
  )

  expr <- expr %>%
    dplyr::left_join(meta, by = "sample")

  expr$gene <- factor(expr$gene, levels = names(available_named))
  expr
}

format_fdr_for_plot <- function(x) {
  out <- rep(NA_character_, length(x))
  keep <- !is.na(x) & x < 0.1

  out[keep & x < 0.001] <- "<0.001"

  idx <- keep & x >= 0.001
  if (any(idx)) {
    formatted <- formatC(x[idx], format = "f", digits = 4)
    formatted <- sub("0+$", "", formatted)
    formatted <- sub("\\.$", "", formatted)
    out[idx] <- formatted
  }

  out
}

prepare_fdr_annotations <- function(stats, box_data) {
  plotted_genes <- unique(as.character(box_data$gene))

  stats <- stats %>%
    dplyr::filter(gene %in% plotted_genes, !is.na(FDR), FDR < 0.1) %>%
    dplyr::mutate(
      gene = as.character(gene),
      label = format_fdr_for_plot(FDR)
    )

  if (nrow(stats) == 0) {
    stats$y.position <- numeric(0)
    return(stats)
  }

  y_positions <- box_data %>%
    dplyr::mutate(gene = as.character(gene)) %>%
    dplyr::group_by(gene) %>%
    dplyr::summarise(
      ymin = min(logCPM, na.rm = TRUE),
      ymax = max(logCPM, na.rm = TRUE),
      yrange = ymax - ymin,
      .groups = "drop"
    )

  stats <- stats %>%
    dplyr::left_join(y_positions, by = "gene") %>%
    dplyr::group_by(gene) %>%
    dplyr::mutate(
      annotation_index = dplyr::row_number() - 1,
      y.position = ymax + dplyr::if_else(
        yrange > 0,
        (0.12 + 0.12 * annotation_index) * yrange,
        0.2 + 0.2 * annotation_index
      )
    ) %>%
    dplyr::ungroup()

  stats$gene <- factor(stats$gene, levels = levels(box_data$gene))
  stats
}

  }

  p
