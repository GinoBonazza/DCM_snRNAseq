pct_expressing <- function(cell_vec) {
  m <- counts[goi, cell_vec, drop = FALSE]
  (Matrix::rowSums(m > 0) / ncol(m)) * 100
}

    dplyr::mutate(
      variable = xvar,
      p_adj_BH = p.adjust(p_value, method = "BH")

# Correlations for DCM only (by cell type), per cell_type × gene
safe_cor_ct <- function(df, xvar) {
  df %>%
    dplyr::filter(!is.na(.data[[xvar]]), !is.na(logCPM)) %>%
    dplyr::group_by(cell_type, gene) %>%
    dplyr::summarise(
      n        = dplyr::n(),
      r        = suppressWarnings(cor(logCPM, .data[[xvar]], method = "pearson")),
      p_value  = tryCatch(cor.test(logCPM, .data[[xvar]], method = "pearson")$p.value,
                          error = function(e) NA_real_),
      .groups = "drop"
    ) %>%
    dplyr::mutate(
      variable = xvar,
      p_adj_BH = p.adjust(p_value, method = "BH")
    )
}
