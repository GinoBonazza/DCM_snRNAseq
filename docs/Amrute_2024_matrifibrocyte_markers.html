Helper functions to read 10x-style MTX files

The dataset is CITE-seq, but this script keeps only RNA / gene-expression rows. If the features.tsv.gz file contains a feature-type column, the script keeps rows annotated as Gene Expression; otherwise it keeps all rows.

read_one_gse217494_sample <- function(sample_id, data_dir, cell_meta) {
  mtx_file <- file.path(data_dir, paste0("GSE217494_", sample_id, ".matrix.mtx.gz"))
  feature_file <- file.path(data_dir, paste0("GSE217494_", sample_id, ".features.tsv.gz"))
  barcode_file <- file.path(data_dir, paste0("GSE217494_", sample_id, ".barcodes.tsv.gz"))

  if (!file.exists(mtx_file) || !file.exists(feature_file) || !file.exists(barcode_file)) {
    stop("Missing one or more files for ", sample_id)
  }

  message("Reading ", sample_id)

  mat <- Matrix::readMM(mtx_file)
  mat <- methods::as(mat, "dgCMatrix")

  features <- readr::read_tsv(feature_file, col_names = FALSE, show_col_types = FALSE)
  barcodes <- readr::read_tsv(barcode_file, col_names = FALSE, show_col_types = FALSE)[[1]]

  if (nrow(features) != nrow(mat)) {
    stop(sample_id, ": number of features does not match matrix rows.")
  }

  if (length(barcodes) != ncol(mat)) {
    stop(sample_id, ": number of barcodes does not match matrix columns.")
  }

  if (ncol(features) >= 3 && any(features[[3]] == "Gene Expression")) {
    rna_rows <- which(features[[3]] == "Gene Expression")
  } else {
    rna_rows <- seq_len(nrow(features))
  }

  gene_symbols <- if (ncol(features) >= 2) features[[2]] else features[[1]]
  gene_symbols <- as.character(gene_symbols)
  gene_symbols[is.na(gene_symbols) | gene_symbols == ""] <- as.character(features[[1]][is.na(gene_symbols) | gene_symbols == ""])

  mat <- mat[rna_rows, , drop = FALSE]
  rownames(mat) <- make.unique(gene_symbols[rna_rows])

  barcodes <- as.character(barcodes)
  bc_for_metadata <- ifelse(
    grepl("-1$", barcodes),
    sub("-1$", paste0("-", sample_id), barcodes),
    paste0(barcodes, "-", sample_id)
  )

  colnames(mat) <- bc_for_metadata

  keep_cells <- intersect(colnames(mat), cell_meta$bc)
  if (length(keep_cells) == 0) {
    warning(sample_id, ": no barcodes matched metadata.")
    return(NULL)
  }

  mat[, keep_cells, drop = FALSE]
}

get_p_label <- function(p) {
  if (is.na(p)) return("p = NA")
  if (p < 0.001) return("p < 0.001")
  paste0("p = ", formatC(p, format = "f", digits = 3))
}

get_fdr_label <- function(p) {
  if (is.na(p)) return(NA_character_)
  if (p < 0.001) return("FDR < 0.001")
  paste0("FDR = ", formatC(p, format = "f", digits = 3))
}
