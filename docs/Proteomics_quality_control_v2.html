Overview

This workflow evaluates the quality of the Olink and LC-MS/MS proteomic datasets before and after sample exclusion. Healthy control samples are excluded from the initial analysis, which includes all DCM and ICM samples.

P11, PK61 and PK63 are subsequently excluded because of poor proteomic data quality identified in the diagnostic plots. PK69 is excluded because duplicate sample identifiers prevent unambiguous patient assignment. ICM samples and female patients are excluded because the downstream analysis focuses on male DCM patients.

Heatmaps include all usable proteins after removal of rows with insufficient observations or zero variance. Missing LC-MS/MS values are imputed only for visualisation. PCA is performed on proteins observed in at least 50% of samples and uses row-median imputation.

make_group_palette <- function(values) {
  group_colours <- c(
    DCM = "#A63A2A",
    ICM = "#004C8C"
  )

  levels_present <- sort(
    unique(
      as.character(
        values[
          !is.na(values)
        ]
      )
    )
  )

  unexpected_levels <- setdiff(
    levels_present,
    names(group_colours)
  )

  if (length(unexpected_levels) > 0L) {
    stop(
      "Unexpected Group label(s): ",
      paste(
        unexpected_levels,
        collapse = ", "
      )
    )
  }

  group_colours[
    levels_present
  ]
}


safe_continuous_colour <- function(
    values,
    colours = c(
      "grey90",
      "grey60",
      "grey20"
    )
) {
  finite_values <- values[
    is.finite(values)
  ]

  if (length(finite_values) == 0L) {
    return(NULL)
  }

  breaks <- c(
    min(finite_values),
    stats::median(finite_values),
    max(finite_values)
  )

  if (length(unique(breaks)) < 3L) {
    spread <- max(
      abs(breaks[1L]) * 0.01,
      0.01
    )

    breaks <- c(
      breaks[1L] - spread,
      breaks[1L],
      breaks[1L] + spread
    )
  }

  circlize::colorRamp2(
    breaks,
    colours
  )
}


impute_rows_median <- function(mat) {
  output <- as.matrix(mat)

  for (i in seq_len(nrow(output))) {
    missing <- !is.finite(
      output[i, ]
    )

    if (!any(missing)) {
      next
    }

    observed <- output[
      i,
      !missing
    ]

    replacement <- if (
      length(observed) > 0L
    ) {
      stats::median(
        observed,
        na.rm = TRUE
      )
    } else {
      0
    }

    output[
      i,
      missing
    ] <- replacement
  }

  output[
    !is.finite(output)
  ] <- 0

  output
}


impute_rows_low_value <- function(mat) {
  output <- as.matrix(mat)

  global_floor <- suppressWarnings(
    stats::quantile(
      output,
      probs = 0.01,
      na.rm = TRUE
    )
  )

  if (!is.finite(global_floor)) {
    global_floor <- 0
  }

  for (i in seq_len(nrow(output))) {
    missing <- !is.finite(
      output[i, ]
    )

    if (!any(missing)) {
      next
    }

    observed <- output[
      i,
      !missing
    ]

    replacement <- if (
      length(observed) > 0L
    ) {
      minimum <- min(
        observed,
        na.rm = TRUE
      )

      spread <- stats::sd(
        observed,
        na.rm = TRUE
      )

      if (
        !is.finite(spread) ||
        spread == 0
      ) {
        spread <- 0.5
      }

      minimum - 0.5 * spread
    } else {
      global_floor
    }

    output[
      i,
      missing
    ] <- replacement
  }

  output[
    !is.finite(output)
  ] <- global_floor

  output
}


filter_usable_features <- function(
    mat,
    minimum_observed = 2L
) {
  mat <- as.matrix(mat)

  observed <- matrixStats::rowSums2(
    is.finite(mat)
  )

  variance <- matrixStats::rowVars(
    mat,
    na.rm = TRUE
  )

  keep <- observed >= minimum_observed &
    is.finite(variance) &
    variance > 0

  mat[
    keep,
    ,
    drop = FALSE
  ]
}


prepare_pca <- function(
    mat,
    metadata,
    assay_name,
    minimum_observed_fraction =
      pca_min_observed_fraction
) {
  mat <- as.matrix(mat)

  observed_fraction <- matrixStats::rowMeans2(
    is.finite(mat)
  )

  variance <- matrixStats::rowVars(
    mat,
    na.rm = TRUE
  )

  keep <- observed_fraction >=
    minimum_observed_fraction &
    is.finite(variance) &
    variance > 0

  pca_matrix <- mat[
    keep,
    ,
    drop = FALSE
  ]

  if (nrow(pca_matrix) < 2L) {
    stop(
      assay_name,
      ": fewer than two proteins are available for PCA."
    )
  }

  pca_matrix <- impute_rows_median(
    pca_matrix
  )

  pca <- stats::prcomp(
    t(pca_matrix),
    center = TRUE,
    scale. = TRUE
  )

  variance_explained <-
    pca$sdev^2 /
    sum(pca$sdev^2) *
    100

  plot_data <- data.frame(
    Sample = rownames(pca$x),
    PC1 = pca$x[, 1L],
    PC2 = pca$x[, 2L],
    stringsAsFactors = FALSE
  ) %>%
    dplyr::left_join(
      metadata,
      by = "Sample"
    )

  list(
    pca = pca,
    variance_explained =
      variance_explained,
    plot_data = plot_data,
    number_features =
      nrow(pca_matrix),
    assay_name = assay_name
  )
}

make_sample_annotation <- function(
    metadata,
    samples,
    include_olink_technical = FALSE
) {
  ordered_metadata <- metadata %>%
    dplyr::slice(
      match(
        samples,
        Sample
      )
    )

  annotation_data <- data.frame(
    Group = factor(
      ordered_metadata$Group
    ),
    row.names = samples,
    check.names = FALSE
  )

  annotation_colours <- list(
    Group = make_group_palette(
      annotation_data$Group
    )
  )

  if (
    isTRUE(include_olink_technical) &&
    "Batch" %in% colnames(ordered_metadata)
  ) {
    annotation_data$Batch <- factor(
      ordered_metadata$Batch
    )

    batch_levels <- levels(
      droplevels(
        annotation_data$Batch
      )
    )

    if (length(batch_levels) > 0L) {
      batch_colours <- setNames(
        scales::hue_pal(
          h = c(30, 300),
          c = 80,
          l = 55
        )(
          length(batch_levels)
        ),
        batch_levels
      )

      annotation_colours$Batch <-
        batch_colours
    }
  }

  if (
    isTRUE(include_olink_technical) &&
    "BCA_mg_ml" %in% colnames(
      ordered_metadata
    )
  ) {
    bca <- as.numeric(
      ordered_metadata$BCA_mg_ml
    )

    if (any(is.finite(bca))) {
      annotation_data$BCA_mg_ml <- bca

      annotation_colours$BCA_mg_ml <-
        safe_continuous_colour(bca)
    }
  }

  ComplexHeatmap::HeatmapAnnotation(
    df = annotation_data,
    col = annotation_colours,
    na_col = "#E5E7EB"
  )
}


draw_expression_heatmap <- function(
    mat,
    metadata,
    assay_name,
    title_suffix,

      "median",
      "low"
    ),
    include_olink_technical = FALSE

    imputation

calculate_sample_metrics <- function(
    mat,
    metadata,
    assay_name
) {
  mat <- as.matrix(mat)

  metrics <- tibble::tibble(
    Sample = colnames(mat),
    number_detected =
      matrixStats::colSums2(
        is.finite(mat)
      ),
    missing_fraction =
      matrixStats::colMeans2(
        !is.finite(mat)
      ),
    median_abundance =
      matrixStats::colMedians(
        mat,
        na.rm = TRUE
      ),
    mean_abundance =
      matrixStats::colMeans2(

    dplyr::left_join(
      metadata %>%
        dplyr::select(
          Sample,
          Group
        ),
      by = "Sample"
    ) %>%
    dplyr::mutate(
      assay = assay_name
    )

  metrics
}

prepare_qc_set <- function(
    excluded_samples,
    label
) {
  clinical_use <- clinical_non_control %>%
    dplyr::filter(
      !Sample %in%
        excluded_samples
    )

  olink_samples <- clinical_use$Sample[
    clinical_use$Sample %in%
      colnames(olink_expression_all)
  ]

  ms_samples <- clinical_use$Sample[
    clinical_use$Sample %in%
      colnames(ms_expression_log2_all)
  ]

  olink_matrix <- olink_expression_all[
    ,
    olink_samples,
    drop = FALSE
  ]

  ms_matrix <- ms_expression_log2_all[
    ,
    ms_samples,
    drop = FALSE
  ]

  olink_metadata <- clinical_use %>%
    dplyr::left_join(
      olink_sample_metadata %>%
        dplyr::select(
          Sample,
          Batch,
          BCA_mg_ml
        ),
      by = "Sample"
    ) %>%
    dplyr::filter(
      Sample %in%
        olink_samples
    ) %>%
    dplyr::slice(
      match(
        olink_samples,
        Sample
      )
    )

  ms_metadata <- clinical_use %>%
    dplyr::filter(
      Sample %in%
        ms_samples
    ) %>%
    dplyr::slice(
      match(
        ms_samples,
        Sample
      )
    )

  availability <- clinical_use %>%
    dplyr::transmute(
      Sample,
      Group,
      has_Olink =
        Sample %in%
        colnames(
          olink_expression_all
        ),
      has_LCMS =
        Sample %in%
        colnames(
          ms_expression_log2_all
        )
    ) %>%
    dplyr::mutate(
      availability = dplyr::case_when(
        has_Olink &
          has_LCMS ~
          "Both Olink and LC-MS/MS",
        has_Olink &
          !has_LCMS ~
          "Olink only",
        !has_Olink &
          has_LCMS ~
          "LC-MS/MS only",
        TRUE ~ "Neither"
      )
    ) %>%
    dplyr::arrange(
      availability,
      Group,
      Sample
    )

  availability_summary <- availability %>%
    dplyr::count(
      availability,
      name = "number_samples"
    ) %>%
    tidyr::complete(
      availability = c(
        "Both Olink and LC-MS/MS",
        "Olink only",
        "LC-MS/MS only",
        "Neither"
      ),
      fill = list(
        number_samples = 0L
      )
    ) %>%
    dplyr::bind_rows(
      tibble::tibble(
        availability =
          "Total non-control samples",
        number_samples =
          nrow(availability)
      )
    )

  list(
    label = label,
    excluded_samples =
      excluded_samples,
    clinical = clinical_use,
    olink = olink_matrix,
    ms = ms_matrix,
    olink_metadata =
      olink_metadata,
    ms_metadata =
      ms_metadata,
    availability =
      availability,
    availability_summary =
      availability_summary
  )
}

Variance partitioning

The contribution of clinical and technical variables to protein abundance variation was estimated separately for the Olink and LC-MS/MS datasets.

run_proteomics_variance_partition <- function(
    expression_matrix,
    metadata,
    candidate_variables
) {
  variables <-
    intersect(
      candidate_variables,
      colnames(metadata)
    )

  analysis_metadata <-
    metadata %>%
    dplyr::select(
      Sample,
      dplyr::all_of(
        variables
      )
    ) %>%
    dplyr::distinct(
      Sample,
      .keep_all = TRUE
    ) %>%
    dplyr::filter(
      Sample %in%
        colnames(
          expression_matrix
        )
    ) %>%
    dplyr::mutate(
      dplyr::across(
        dplyr::all_of(
          variables
        ),
        as.numeric
      )
    ) %>%
    tidyr::drop_na()

  variables <-
    variables[
      vapply(
        analysis_metadata[
          variables
        ],
        function(values) {
          is.finite(
            stats::sd(values)
          ) &&
            stats::sd(values) > 0
        },
        logical(1)
      )
    ]

  analysis_metadata <-
    analysis_metadata %>%
    dplyr::select(
      Sample,
      dplyr::all_of(
        variables
      )
    )

  if (
    length(variables) == 0L
  ) {
    stop(
      "No usable variables were available for variance partitioning."
    )
  }

  if (
    nrow(analysis_metadata) <=
      length(variables) + 2L
  ) {
    stop(
      "Too few complete samples for the selected variance-partition model."
    )
  }

  analysis_metadata <-
    analysis_metadata %>%
    dplyr::mutate(
      dplyr::across(
        dplyr::all_of(
          variables
        ),
        ~ as.numeric(
          scale(.x)
        )
      )
    )

  analysis_matrix <-
    expression_matrix[
      ,
      analysis_metadata$Sample,
      drop = FALSE
    ]

  observed_values <-
    matrixStats::rowSums2(
      is.finite(
        analysis_matrix
      )
    )

  feature_variance <-
    matrixStats::rowVars(
      analysis_matrix,
      na.rm = TRUE
    )

  usable_features <-
    observed_values >= 3L &
    is.finite(feature_variance) &
    feature_variance > 0

  analysis_matrix <-
    analysis_matrix[
      usable_features,
      ,
      drop = FALSE
    ]

  variance_formula <-
    stats::reformulate(
      variables
    )

  variance_partition <-
    variancePartition::fitExtractVarPartModel(
      analysis_matrix,
      variance_formula,
      analysis_metadata
    )

  colnames(
    variance_partition
  ) <-
    gsub(
      "`",
      "",
      colnames(
        variance_partition
      ),
      fixed = TRUE
    )

  mean_variance <-
    sort(
      colMeans(
        variance_partition,
        na.rm = TRUE
      ),
      decreasing = TRUE
    )

  plot_data <-
    tibble::tibble(
      variable =
        names(
          mean_variance
        ),
      mean_fraction =
        as.numeric(
          mean_variance
        )
    ) %>%
    dplyr::mutate(
      variable_label =
        dplyr::recode(
          variable,
          !!!clinical_variable_labels,
          .default =
            variable
        ),
      variable_label =
        factor(
          variable_label,
          levels =
            rev(
              variable_label
            )
        )
    )

  list(
    variance_partition =
      variance_partition,
    plot_data =
      plot_data,
    metadata =
      analysis_metadata,
    number_features =
      nrow(
        analysis_matrix
      )
  )
}
