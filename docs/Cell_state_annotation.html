set_donor_age <- function(x) {
    is_seurat <- inherits(x, "Seurat")
    md <- if (is_seurat) x[[]] else x
    if (is.data.frame(md) && all(c("Sample", "Age_years") %in% names(md))) {
        j <- match(trimws(as.character(md$Sample)), donor_ages$Sample)
        present <- !is.na(md$Sample) & nzchar(trimws(as.character(md$Sample)))
        stopifnot(!anyNA(j[present]))
        x$Age_years <- donor_ages$Age_years[j]
    }
    x
}

Combine cell states

rna <- set_donor_age(qs2::qs_read(here::here("data", "processed", "RNA_quality_control.qs2"), nthreads = 2))
lineages <- c(CM = "Cardiomyocytes", FB = "Fibroblasts", EC = "Endothelial cells", MC = "Mural cells", MP = "Macrophages", 
    LC = "Lymphocytes")
states <- list()
retained <- list()
before <- list()
for (tag in names(lineages)) {
    object <- set_donor_age(qs2::qs_read(here::here("data", "processed", tag, "annotated.qs2"), nthreads = 2))
    initial <- set_donor_age(qs2::qs_read(here::here("data", "processed", tag, "baseline.qs2"), nthreads = 2))
    before[[tag]] <- colnames(initial)
    retained[[tag]] <- colnames(object)
    states[[tag]] <- data.frame(cell = colnames(object), cell_state = as.character(object$cell_state), Sample = object$Sample, 
        cell_type = object$cell_type)
}
annotation <- bind_rows(states)
stopifnot(!anyDuplicated(annotation$cell), all(annotation$cell %in% colnames(rna)))
removed <- setdiff(unlist(before, use.names = FALSE), unlist(retained, use.names = FALSE))
removal_summary <- rna[[]][removed, , drop = FALSE] %>% count(Group, cell_type, Sample, name = "Removed_nuclei")
rna <- subset(rna, cells = setdiff(colnames(rna), removed))
rna$cell_state <- annotation$cell_state[match(colnames(rna), annotation$cell)]
expected <- set_donor_age(qs2::qs_read(here::here("data", "processed", "RV_cell_states_RNA.qs2"), nthreads = 2))
stopifnot(setequal(colnames(rna), colnames(expected)))
stopifnot(identical(as.character(rna$cell_state), as.character(expected$cell_state[match(colnames(rna), colnames(expected))])))
stopifnot(isTRUE(all.equal(GetAssayData(rna, assay = "RNA", slot = "counts"), GetAssayData(expected, assay = "RNA", 
    slot = "counts")[, colnames(rna), drop = FALSE], tolerance = 0)))
qs2::qs_save(rna, file.path(output_dir_data, "RV_cell_states_RNA.qs2"), nthreads = 2)
write.csv(annotation, file.path(output_dir_data, "Cell_annotations.csv"), row.names = FALSE)
write.csv(removal_summary, file.path(output_dir_data, "Removed_nuclei_by_sample.csv"), row.names = FALSE)
