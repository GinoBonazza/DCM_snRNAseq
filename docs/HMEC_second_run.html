ep_str_wrap <- function(string, width) {
    x <- gregexpr(' ', string)
    vapply(seq_along(x),
           FUN = function(i) {
               y <- x[[i]]
               n <- nchar(string[i])
               len <- (c(y,n) - c(0, y)) ## length + 1
               idx <- len > width
               j <- which(!idx)
               if (length(j) && max(j) == length(len)) {
                   j <- j[-length(j)]
               }
               if (length(j)) {
                   idx[j] <- len[j] + len[j+1] > width
               }
               idx <- idx[-length(idx)] ## length - 1
               start <- c(1, y[idx] + 1)
               end <- c(y[idx] - 1, n)
               words <- substring(string[i], start, end)
               paste0(words, collapse="\n")
           },
           FUN.VALUE = character(1)
    )
}

plot_program_score <- function(df, score_col, title) {
  df <- df %>%
    dplyr::select(Sample, Group, score = dplyr::all_of(score_col)) %>%
    dplyr::filter(!is.na(score))

  fit <- lm(score ~ Group, data = df)
  p_group <- broom::tidy(fit) %>%
    dplyr::filter(term == "GroupHF") %>%
    dplyr::pull(p.value)

  y_max <- max(df$score, na.rm = TRUE)
  y_min <- min(df$score, na.rm = TRUE)
  y_range <- y_max - y_min
  if (!is.finite(y_range) || y_range == 0) y_range <- 1

  ggplot(df, aes(x = Group, y = score, fill = Group)) +
    geom_boxplot(width = 0.55, outlier.shape = NA, color = "black", linewidth = 0.5) +
    geom_jitter(width = 0.10, size = 1.8, shape = 21, stroke = 0.4, color = "black") +
    geom_segment(
      aes(x = 1, xend = 2, y = y_max + 0.10 * y_range, yend = y_max + 0.10 * y_range),
      inherit.aes = FALSE
    ) +
    annotate(
      "text",
      x = 1.5,
      y = y_max + 0.16 * y_range,
      label = format_p_label(p_group),
      size = 5,
      fontface = format_p_fontface(p_group)
    ) +
    scale_fill_manual(values = group_cols) +
    coord_cartesian(ylim = c(y_min - 0.05 * y_range, y_max + 0.22 * y_range)) +
    labs(title = title, x = NULL, y = "Score") +
    theme_classic(base_size = 12) +
    theme(
      plot.title = element_text(face = "bold", hjust = 0.5),
      axis.text.x = element_text(angle = 45, hjust = 1, face = "bold"),
      axis.title.y = element_text(face = "bold"),
      legend.position = "none"
    )
}

plot_gene_logcpm <- function(gene_name) {
  idx <- which(rownames(logcpm_scores_gene) == gene_name)

  if (length(idx) == 0) {
    stop("Gene not found: ", gene_name)
  }

  gene_mat <- logcpm_scores_gene[idx, , drop = FALSE]

  if (nrow(gene_mat) > 1) {
    gene_values <- colMeans(gene_mat, na.rm = TRUE)
  } else {
    gene_values <- as.numeric(gene_mat[1, ])
  }

  if (length(gene_values) != ncol(logcpm_scores_gene)) {
    stop(
      "Length mismatch for ", gene_name,
      ": got ", length(gene_values),
      " values, expected ", ncol(logcpm_scores_gene)
    )
  }

  df <- tibble::tibble(
    Sample = colnames(logcpm_scores_gene),
    gene = rep(gene_name, ncol(logcpm_scores_gene)),
    logCPM = gene_values,
    Group = factor(meta[colnames(logcpm_scores_gene), "Group", drop = TRUE], levels = c("HC", "HF"))
  )

  fit <- lm(logCPM ~ Group, data = df)

  p_group <- broom::tidy(fit) %>%
    dplyr::filter(term == "GroupHF") %>%
    dplyr::pull(p.value)

  y_max <- max(df$logCPM, na.rm = TRUE)
  y_min <- min(df$logCPM, na.rm = TRUE)
  y_range <- y_max - y_min
  if (!is.finite(y_range) || y_range == 0) y_range <- 1

  ggplot(df, aes(x = Group, y = logCPM, fill = Group)) +
    geom_boxplot(width = 0.55, outlier.shape = NA, color = "black", linewidth = 0.5) +
    geom_jitter(width = 0.10, size = 1.8, shape = 21, stroke = 0.4, color = "black") +
    geom_segment(
      aes(x = 1, xend = 2, y = y_max + 0.10 * y_range, yend = y_max + 0.10 * y_range),
